\begin{enumerate}[itemsep=3em]
    \item
    اكتب برنامجًا يستقبل من المستخدم عددًا صحيحًا من 3 منازل (مثلاً 472).
    البرنامج يقوم باستخراج منازل العدد (الآحاد، العشرات، والمئات) ويطبع كل منها على سطر منفصل.
    \ifdetailed
    \begin{boxExample}
    \begin{english}
    \begin{minted}{csharp}
Please enter a 3-digit number:
472
Units: 2
Tens: 7
Hundreds: 4
    \end{minted}
    \end{english}
    \end{boxExample}
    \ifwithsols
    \begin{boxSolution}
    \begin{english}
    \begin{minted}{csharp}
int num = int.Parse(Console.ReadLine());

int units = num % 10;
num = num / 10;
int tens = num % 10;
int hundreds = num / 10;

Console.WriteLine("Units: " + units);
Console.WriteLine("Tens: " + tens);
Console.WriteLine("Hundreds: " + hundreds);
    \end{minted}
    \end{english}
    \end{boxSolution}
    \clearpage
    \fi
    \fi

    \item
    اكتب برنامجًا يستقبل من المستخدم عددًا إجماليًا للطلاب في المدرسة.
    البرنامج يقوم بتوزيع الطلاب على صفوف، بحيث يتسع كل صف لـ $30$ طالبًا كحد أقصى.
    البرنامج يطبع عدد الصفوف الممتلئة، وعدد الطلاب المتبقين (الذين لا يكفون لفتح صف ممتلئ).
    \ifdetailed
    \begin{boxExample}
    \begin{english}
    \begin{minted}{csharp}
Please enter the total number of students:
100
Full classes: 3
Remaining students: 10
    \end{minted}
    \end{english}
    \end{boxExample}
    \ifwithsols
    \begin{boxSolution}
    \begin{english}
    \begin{minted}{csharp}
int students = int.Parse(Console.ReadLine());

int fullClasses = students / 30;
int remaining = students % 30;

Console.WriteLine("Full classes: " + fullClasses);
Console.WriteLine("Remaining students: " + remaining);
    \end{minted}
    \end{english}
    \end{boxSolution}
    \fi
    \fi
    \clearpage

    \item
    اكتب برنامجًا يقرأ طولي ضلعي القائمة في مثلث قائم الزاوية $(a, b)$، ثم يحسب ويطبع طول الوتر $c$ باستخدام نظرية فيثاغورس:
    \[
        c = \sqrt{a^2 + b^2}
    \]
    \ifdetailed
    \begin{boxExample}
        \begin{english}
    \begin{minted}{csharp}
Please enter the lengths of the two legs:
3
4
The hypotenuse = 5
    \end{minted}
\end{english}
\end{boxExample}
\ifwithsols
\begin{boxSolution}
    \begin{english}
        \begin{minted}{csharp}
double a = double.Parse(Console.ReadLine());
double b = double.Parse(Console.ReadLine());
double c = Math.Sqrt(Math.Pow(a, 2) + Math.Pow(b, 2));
Console.WriteLine("The hypotenuse = " + c);
        \end{minted}
    \end{english}
\end{boxSolution}
\clearpage
\fi
\fi


    \item
    اكتب برنامجًا يستقبل من المستخدم أطوال أضلاع مثلث قائم الزاوية ($a$ و $b$ و $c$).
    البرنامج يحسب مساحة هذا المثلث ويطبعها. \\
    \textbf{ملاحظة:} في المثلث قائم الزاوية، الوتر هو أطول ضلع، والضلعان الآخران هما القائمان. والمساحة تساوي نصف حاصل ضرب القائمين.
    \ifdetailed
    \begin{boxExample}
    \begin{english}
    \begin{minted}{csharp}
Please enter the 3 side lengths:
3
5
4
Area = 6
    \end{minted}
    \end{english}
    \end{boxExample}
    \ifwithsols
    \begin{boxSolution}
    \begin{english}
    \begin{minted}{csharp}
double a = double.Parse(Console.ReadLine());
double b = double.Parse(Console.ReadLine());
double c = double.Parse(Console.ReadLine());

double maxAB = Math.Max(a, b);
double hypotenuse = Math.Max(maxAB, c);

// The product of the two legs is (a * b * c) / hypotenuse
double productOfLegs = (a * b * c) / hypotenuse;
double area = productOfLegs / 2.0;

Console.WriteLine("Area = " + area);
    \end{minted}
    \end{english}
    \end{boxSolution}
    \fi
    \fi
    \clearpage

    \item
    اكتب برنامجًا يستقبل من المستخدم عددًا عشريًّا يمثّل قيمة $x$.
    البرنامج يقوم بحساب وطباعة قيمة الدالة التالية عند النقطة $x$:
    $$ f(x) = |x - 5| + x^4 $$
    \ifdetailed
    \begin{boxExample}
    \begin{english}
    \begin{minted}{csharp}
Please enter x:
-2
f(x) = 23
    \end{minted}
    \end{english}
    \end{boxExample}
    \ifwithsols
    \begin{boxSolution}
    \begin{english}
    \begin{minted}{csharp}
double x = double.Parse(Console.ReadLine());

double absolute = Math.Abs(x - 5);
double power = Math.Pow(x, 4);
double result = absolute + power;

Console.WriteLine("f(x) = " + result);
    \end{minted}
    \end{english}
    \end{boxSolution}
    \clearpage
    \fi
    \fi

    \item \textbf{سؤال بونوس:}
    اكتب برنامجًا يستقبل من المستخدم معاملات معادلة تربيعية:
    \[
        ax^2 + bx + c = 0
    \]
    واطبع له الحلول باستخدام الدستور:
    \[
        x = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}
    \]
    (\textbf{ملاحظة} افترض أنّ للمعادلة حلّين، وما تحت الجذر موجب.)
    \ifdetailed
    \begin{boxExample}
        \begin{english}
            \begin{minted}{csharp}
Please enter a, b, c:
1
-3
2
The solutions are: x1 = 2, x2 = 1
\end{minted}
\end{english}
\end{boxExample}
\ifwithsols
\begin{boxSolution}
    \begin{english}
    \begin{minted}{csharp}
double a = double.Parse(Console.ReadLine());
double b = double.Parse(Console.ReadLine());
double c = double.Parse(Console.ReadLine());

double delta = Math.Pow(b, 2) - 4 * a * c;
double x1 = (-b + Math.Sqrt(delta)) / (2 * a);
double x2 = (-b - Math.Sqrt(delta)) / (2 * a);
Console.WriteLine("The solutions are: x1 = " + x1 + ", x2 = " + x2);
            \end{minted}
        \end{english}
    \end{boxSolution}
    \fi
    \fi

\end{enumerate}
