    \textbf{مثال:} إذا تلقّت العدد $23481$ فإنّها تعيد $8$.
    \item اكتب برنامجًا يستقبل من المستخدم $100$ عدد صحيح موجب، وفي النهاية يطبع العدد الذي ظهرت فيه أكبر خانة. إذا كان هناك أكثر من عدد ظهرت فيه أكبر خانة، يطبع البرنامج العدد الأكبر من بينها.

\begin{boxSolution}[فرع أ]
\begin{english}

public static int MaxDigit(int n)
{
    if (n == 0)
    {
        return 0;
    }
    int md = 0;
    while (n > 0)
    {
        md = Math.Max(md, n % 10);
        n /= 10;
    }
    return md;
}

\end{english}
\end{boxSolution}

\item اكتب عملية خارجية باسم \textenglish{DigitExists} تتلقى عددًا صحيحًا غير سالب $n$ ورقمًا مكوّنًا من منزلة واحدة $d$ ($0 \le d \le 9$). \\
تعيد العملية \textenglish{true} إذا كان الرقم $d$ يظهر كإحدى خانات العدد $n$، وتعيد \textenglish{false} خلاف ذلك. \\
\textbf{مثال:} استدعاء العملية مع العدد $4729$ والرقم $7$ يعيد \textenglish{true}، بينما استدعاؤها مع العدد $4729$ والرقم $5$ يعيد \textenglish{false}.
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static bool DigitExists(int n, int d)
{
    if (n == 0 && d == 0)
    {
        return true;
    }
    while (n > 0)
    {
        if (n % 10 == d)
        {
            return true;
        }
        n /= 10;
    }
    return false;
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item اكتب عملية خارجية باسم \textenglish{AreDigitsAscending} تتلقى عددًا صحيحًا موجبًا $n$، وتعيد \textenglish{true} إذا كانت خانات العدد مرتّبة تصاعديًّا تمامًا من اليسار إلى اليمين (أي أنّ كل خانة أكبر تمامًا من الخانة التي على يسارها)، وتعيد \textenglish{false} خلاف ذلك. \\
\textbf{مثال:} للعدد $1359$ وللعدد $7$ تعيد العملية \textenglish{true}، بينما للعدد $1428$ وللعدد $1225$ تعيد \textenglish{false}.
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static bool AreDigitsAscending(int n)
{
    while (n >= 10)
    {
        int rightDigit = n % 10;
        int leftDigit = (n / 10) % 10;
        if (rightDigit <= leftDigit)
        {
            return false;
        }
        n /= 10;
    }
    return true;
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item
\begin{enumerate}
    \item اكتب عملية خارجية باسم \textenglish{SumSquares} تتلقى عددًا صحيحًا موجبًا، وتعيد مجموع \textbf{تربيع} خاناته. \\
    \textbf{مثال:} إذا تلقّت العملية العدد $123$، فإنّها تعيد: $1^2 + 2^2 + 3^2 = 1 + 4 + 9 = 14$.
    \item في البرنامج الرئيسي، استدعِ العملية مع العدد $23$ واطبع النتيجة.
    \item اكتب مقطع برنامج في \textenglish{Main} يطبع جميع الأعداد بين $1$ و $10000$ التي تساوي مجموع تربيع خاناتها.
\end{enumerate}
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static int SumSquares(int n)
{
    int sum = 0;
    while (n > 0)
    {
        int d = n % 10;
        sum += d * d;
        n /= 10;
    }
    return sum;
}

public static void Main(string[] args)
{
    Console.WriteLine(SumSquares(23));

    for (int i = 1; i <= 10000; i++)
    {
        if (SumSquares(i) == i)
        {
            Console.WriteLine(i);
        }
    }
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item نعرّف «عددًا مميّزًا» بأنّه عدد موجب إذا رفعنا كل خانة من خاناته لقوة مساوية لعدد خاناته ثم جمعنا النتائج فإنّنا نحصل على العدد الأصلي نفسه. \\
\textbf{مثال:} العدد $153$ هو عدد مميّز، لأنّه مكوّن من $3$ خانات، و:
$1^3 + 5^3 + 3^3 = 1 + 125 + 27 = 153$.
\begin{enumerate}
    \item اكتب عملية خارجية باسم \textenglish{IsSpecial} تتلقى عددًا صحيحًا موجبًا، وتعيد \textenglish{true} إذا كان العدد مميّزًا، و \textenglish{false} خلاف ذلك. \\
    يمكنك الاستعانة بالعملية \textenglish{DigitsCount} من السؤال (\ref{item:digitCount}) لمعرفة عدد خانات العدد.
    \item في البرنامج الرئيسي، اطبع جميع الأعداد المميّزة من $1$ إلى $10000$ بالاستعانة بالعملية \textenglish{IsSpecial}.
\end{enumerate}
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static bool IsSpecial(int n)
{
    int k = DigitsCount(n);
    int m = n, sum = 0;
    while (m > 0)
    {
        int d = m % 10;
        sum += (int)Math.Pow(d, k);
        m /= 10;
    }
    if (sum == n)
    {
        return true;
    }
    else
    {
        return false;
    }
}

public static void Main(string[] args)
{
    for (int i = 1; i <= 10000; i++)
    {
        if (IsSpecial(i))
        {
            Console.WriteLine(i);
        }
    }
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

% --- هـ: بناء عدد، عكس عدد، والعدد المتماثل (Palindrome) ---
\item
\begin{enumerate}
    \item اكتب عملية خارجية باسم \textenglish{AppendDigit} تتلقى عددًا صحيحًا غير سالب، ورقمًا مكوّنًا من منزلة واحدة. \\
    العملية تلصق المنزلة في يمين العدد الأوّل وتعيد النتيجة. \\
    \textbf{مثال:} إذا تلقّت العملية العدد $12$ والمنزلة $5$ فإنّها تعيد $125$.
    \item اكتب برنامجًا يستقبل من المستخدم أرقامًا (كل منها مكوّن من منزلة واحدة) ويكوّن عددًا صحيحًا من هذه الأرقام ويطبعه. \\
    ينتهي استقبال الأرقام عندما يُدخل المستخدم عددًا سالبًا. \\
    \textbf{مثال:} إذا أدخل المستخدم الأرقام $5$ ثم $1$ ثم $2$ ثم $4$ ثم $-1$ فعلى البرنامج أن يطبع $5124$. \\
    عليك استخدام العملية \textenglish{AppendDigit} من الفرع السابق (لا حاجة للتحقق من صحة المدخلات).
\end{enumerate}
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static int AppendDigit(int n, int d)
{
    return n * 10 + d;
}

public static void Main(string[] args)
{
    int res = 0;
    int d = int.Parse(Console.ReadLine());
    while (d >= 0)
    {
        res = AppendDigit(res, d);
        d = int.Parse(Console.ReadLine());
    }
    Console.WriteLine(res);
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item
\begin{enumerate}
    \item اكتب عملية خارجية باسم \textenglish{ReverseNumber} تتلقى عددًا صحيحًا غير سالب $n$، وتعيد العدد الناتج من عكس ترتيب خاناته. \\
    \textbf{مثال:} إذا تلقّت العملية العدد $1234$ فإنّها تعيد $4321$.
    \item نسمّي العدد «متماثلاً» (بوليندروم) إذا كان يقرأ من اليمين إلى اليسار كما يقرأ من اليسار إلى اليمين (مثل: $12321$، $4554$، $7$). \\
    اكتب عملية خارجية باسم \textenglish{IsPalindrome} تتلقى عددًا صحيحًا غير سالب، وتعيد \textenglish{true} إذا كان العدد متماثلاً، و \textenglish{false} خلاف ذلك (استعن بالعملية \textenglish{ReverseNumber}).
\end{enumerate}
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
// a
public static int ReverseNumber(int n)
{
    int rev = 0;
    while (n > 0)
    {
        rev = rev * 10 + (n % 10);
        n /= 10;
    }
    return rev;
}

// b
public static bool IsPalindrome(int n)
{
    if (n == ReverseNumber(n))
    {
        return true;
    }
    else
    {
        return false;
    }
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item \label{item:appendDigitToLeft}
اكتب عملية خارجية باسم \textenglish{AppendDigitToLeft} تتلقى عددًا صحيحًا موجبًا، ورقمًا مكوّنًا من منزلة واحدة. \\
العملية تلصق المنزلة بالعدد الأوّل \textbf{من اليسار} وتعيد النتيجة. \\
\textbf{مثال:} إذا تلقّت العملية العدد $12$ والمنزلة $5$ فإنّها تعيد $512$.
\begin{boxHint}
يمكنك الاستعانة بالعملية \textenglish{DigitsCount} التي كتبتها في السؤال (\ref{item:digitCount}) لمعرفة عدد خانات العدد. \\
لكي نحصل على $512$ من $5$ و $12$ نضرب الـ $5$ بـ $100$ (وهو $10$ مرفوعًا لقوة عدد خانات الـ $12$) ثم نجمع الـ $12$.
\end{boxHint}
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static int AppendDigitToLeft(int n, int d)
{
    int dc = DigitsCount(n);
    int p = (int)Math.Pow(10, dc);
    return d * p + n;
}
\end{minted}
\end{english}
\end{boxSolution}
\fi
\filbreak

\end{enumerate}

\section*{القسم الثاني: تحليل العدد إلى عوامله (قواسمه) والأعداد الأوّلية}

\begin{enumerate}[resume, itemsep=2em]

\item \label{item:divisors_sum}
اكتب عملية خارجية باسم \textenglish{DivisorsSum} تتلقى عددًا صحيحًا موجبًا وتعيد \textbf{مجموع} قواسمه \textbf{باستثناء نفسه}. \\
\textbf{مثال:} إذا تلقّت العدد $8$، فإنّها تعيد: $1 + 2 + 4 = 7$.
\ifwithsols
\begin{boxSolution}

public static int DivisorsSum(int n)
{
    int sum = 0;
    for (int d = 1; d < n; d++)
    {
        if (n % d == 0)
        {
            sum += d;
        }
    }
    return sum;
}
\end{minted}
\end{english}
\end{boxSolution}

\item \textbf{العدد الكامل:} هو عدد موجب يساوي مجموع قواسمه باستثناء نفسه. \\

اكتب عملية خارجية باسم \textenglish{IsPerfectNumber} تتلقى عددًا صحيحًا موجبًا وتعيد \textenglish{true} إذا كان عددًا كاملاً، و \textenglish{false} وإلا (يمكنك الاستعانة بالعملية \textenglish{DivisorsSum} من السؤال \ref{item:divisors_sum}).

public static bool IsPerfectNumber(int n)

    for (int i = 1; i < n; i++)

        if (n % i == 0)
        {
            sum += i;

    return sum == n;
}
\end{minted}

\item \textbf{عددان صديقان:} هما عددان موجبان يحقّقان أنّ مجموع قواسم الأوّل (باستثناء نفسه) يساوي العدد الثاني، ومجموع قواسم الثاني (باستثناء نفسه) يساوي العدد الأوّل. \\
\textbf{مثال:} العددان $220$ و $284$ هما عددان صديقان، لأنّ مجموع قواسم $220$ (بدون نفسه) يساوي $284$، ومجموع قواسم $284$ (بدون نفسه) يساوي $220$. \\

public static bool AreAmicable(int a, int b)

\item اكتب عملية خارجية باسم \textenglish{GCD} تتلقى عددين صحيحين موجبين $a$ و $b$، وتعيد القاسم المشترك الأكبر لهما (أكبر عدد يقبل كلا العددين القسمة عليه بدون باقٍ). \\
\textbf{مثال:} القاسم المشترك الأكبر للعددين $12$ و $15$ هو $3$. \\
\textbf{ملاحظة:} القاسم المشترك الأكبر يكون بالضرورة أصغر من أو مساويًا لـ \textenglish{Math.Min(a, b)}.
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static int GCD(int a, int b)
{
    int min = Math.Min(a, b);
    for (int i = min; i >= 1; i--)
    {
        if (a % i == 0 && b % i == 0)
        {
            return i;
        }
    }
    return 1;
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item اكتب عملية خارجية باسم \textenglish{GCDThreeNumbers} تتلقى ثلاثة أعداد صحيحة موجبة $a$ و $b$ و $c$، وتعيد القاسم المشترك الأكبر للأعداد الثلاثة. \\

            return i;
        }
    }
    return 1;
}
\end{minted}
\end{english}

\item اكتب عملية خارجية باسم \textenglish{LCM} تتلقى عددين صحيحين موجبين $a$ و $b$، وتعيد المضاعف المشترك الأصغر لهما (أصغر عدد موجب يقبل القسمة على العددين معًا بدون باقٍ). \\
\textbf{مثال:} المضاعف المشترك الأصغر للعددين $6$ و $8$ هو $24$، لأنّه أصغر عدد موجب يقبل القسمة على $6$ وعلى $8$ معًا. \\
\textbf{ملاحظة:} المضاعف المشترك الأصغر يكون بالضرورة أكبر من أو مساويًا لـ \textenglish{Math.Max(a, b)} وأصغر من أو مساويًا لـ $a \cdot b$.
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static int LCM(int a, int b)
{
    int max = Math.Max(a, b);
    for (int i = max; i < (a * b); i++)
    {
        if (i % a == 0 && i % b == 0)
        {
            return i;
        }
    }
    return a * b;
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item اكتب عملية خارجية باسم \textenglish{MinTermsToExceed} تتلقى عددًا عشريًّا موجبًا $target$، وتجد وتعيد أصغر عدد حدود $n$ يجب جمعها من السلسلة:
\[ 1 + \frac{1}{2} + \frac{1}{3} + \dots + \frac{1}{n} \]
لكي يصبح المجموع أكبر تمامًا من $target$. \\
\textbf{مثال:} إذا تلقّت العملية العدد $1.7$ فإنّها تعيد $3$ (لأنّ $1 + \frac{1}{2} = 1.5 \le 1.7$، بينما $1 + \frac{1}{2} + \frac{1}{3} \approx 1.833 > 1.7$).
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{csharp}
public static int MinTermsToExceed(double target)
{
    double sum = 0;
    int n = 0;
    while (sum <= target)
    {
        n++;
        sum += 1.0 / n;
    }
    return n;
}
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item متوالية فيبوناتشي معرّفة بالشكل التالي:
\begin{english}
\begin{align*}
    a_1 &= 1 \\
    a_2 &= 1 \\
    a_n &= a_{n-1} + a_{n-2} \quad \text{for } n > 2
\end{align*}
\end{english}
مثلاً، أوّل $7$ حدود من متوالية فيبوناتشي هي: $1, 1, 2, 3, 5, 8, 13$.
\begin{enumerate}
    \item اكتب عملية خارجية باسم \textenglish{Fib} تتلقى عددًا صحيحًا موجبًا $n$ وتعيد الحد الـ $n$ من متوالية فيبوناتشي (مثلاً: إذا تلقّت العدد $3$ تعيد $2$، وإذا تلقّت $6$ تعيد $8$).
    \item اكتب عملية خارجية باسم \textenglish{PrintFirstNFibs} تتلقى عددًا صحيحًا موجبًا $n$، وتطبع أوّل $n$ حدود من متوالية فيبوناتشي بالاستعانة بالعملية \textenglish{Fib} من البند (أ). \\
    مثلاً، إذا تلقّت العملية العدد $8$، تطبع الأعداد: $1, 1, 2, 3, 5, 8, 13, 21$ (كل عدد في سطر).
    \item اكتب برنامجًا رئيسيًّا يستقبل عددًا صحيحًا موجبًا من المستخدم، ويستدعي العملية \textenglish{PrintFirstNFibs} مع العدد المدخل.
\end{enumerate}
\ifwithsols
\begin{boxSolution}[فرع أ]
\begin{english}
\begin{minted}{csharp}
public static int Fib(int n)
{
    if (n <= 2)
    {
        return 1;
    }
    int a = 1, b = 1;
    for (int i = 3; i <= n; i++)
    {
        int c = a + b;
        a = b;
        b = c;
    }
    return b;
}
\end{minted}
\end{english}
\end{boxSolution}

public static void PrintFirstNFibs(int n)
{
    for (int i = 1; i <= n; i++)
    {
        Console.WriteLine(Fib(i));
    }
}

public static void Main(string[] args)
{
    int n = int.Parse(Console.ReadLine());
    PrintFirstNFibs(n);
}
\end{minted}
