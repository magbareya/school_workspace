\begin{boxWarning}
\textbf{تعليمات هامة للعمل:}
تعتمد هذه المرحلة على الفئات التي بنيتموها في المراحل السابقة (\textenglish{BasicShip} و \textenglish{ScoutShip} و \textenglish{HibernationShip} و \textenglish{MotherShip}) — حدّثوا الفئات الموجودة ولا تبدأوا من الصفر. \\
في \textbf{كل تمرين}، يجب تحديث مقطع الفحص داخل العملية الرئيسية \textenglish{Main} وتشغيله للتأكد من صحة عمل الفئات والعمليات الجديدة التي كتبتموها.
\end{boxWarning}

% ==============================================================
\section*{المرحلة الثالثة: مركز التحكم وتعدد الأشكال}
% ==============================================================

\begin{enumerate}[label=\textbf{تمرين \arabic*:}, wide=0pt, leftmargin=*, itemsep=2.2em]

    \item \textbf{بناء فئة مركز التحكم (\textenglish{FleetControl})} \\
    ابنوا فئة خدمات باسم \textenglish{FleetControl} (مركز التحكم في الأسطول)، وهي الفئة التي تدير كل مركبات الأسطول:

    {\centering
    \renewcommand{\arraystretch}{1.2}
    \small
    \begin{tabular}{|>{\raggedleft\arraybackslash}p{0.4\textwidth}|>{\raggedright\arraybackslash}p{0.5\textwidth}|}
    \hline
    \multicolumn{2}{|c|}{\textbf{الفئة: \textenglish{FleetControl}}} \\
    \hline
    صفة: مصفوفة تحوي مركبات من كل الأنماط (\textenglish{ScoutShip}، \textenglish{HibernationShip}، \textenglish{MotherShip})، مع عمليتي \textenglish{Get} و \textenglish{Set} لها. & \textenglish{private BasicShip[] ships;} \newline \textenglish{GetShips(), SetShips(BasicShip[] ships)} \\
    \hline
    عملية بنائية تتلقى حجم المصفوفة وتنشئها (في البداية كل الحدود فارغة، أي \textenglish{null}). & \textenglish{public FleetControl(int maxShips)} \\
    \hline
    عملية تتلقى كائناً من النمط \textenglish{BasicShip} (أي مركبة من أي نمط) وتضيفه في \textbf{أول مكان فارغ} في المصفوفة. تعيد \textenglish{true} إذا أُضيفت المركبة و \textenglish{false} إذا كانت المصفوفة ممتلئة أو كان الكائن \textenglish{null}. & \textenglish{Add(BasicShip ship)} \\
    \hline
    عملية تعيد سلسلة نصية تصف كل المركبات الموجودة في المصفوفة (انتبهوا: لا يجوز أن يتعطل البرنامج بسبب الحدود الفارغة \textenglish{null}). & \textenglish{ToString()} \\
    \hline
    \end{tabular}
    \par} %

    \vspace{2em}
    \noindent \textbf{الفحص في \textenglish{Main}:} \\
    أنشئوا مركز تحكم بحجم 6، وأضيفوا إليه مركبات من الأنماط الثلاثة، واطبعوا الوصف. ماذا يحدث عند استدعاء \textenglish{ToString} لكل مركبة؟ لاحظوا أنّ كل مركبة "تعرف" كيف تصف نفسها!

    \item \textbf{إضافة عمليات قابلة لإعادة التعريف (\textenglish{virtual} و \textenglish{override})}
    \begin{enumerate}[label=\textbf{(\arabic*)}]
        \item أضيفوا إلى مشروع الأسطول عملية عامة قابلة لإعادة التعريف تعريفها:
        \begin{center}
        \textenglish{public virtual bool IsInDanger()}
        \end{center}
        \begin{itemize}
            \item تُعرَّف العملية للمرة الأولى في الفئة \textenglish{BasicShip} وتعيد \textenglish{false} دائماً.
            \item في الفئة \textenglish{ScoutShip} يُعاد تعريف العملية (\textenglish{override})، وتعيد \textenglish{true} إذا كانت المركبة \textbf{تستخدم الاحتياطي الطارئ} (الوقود سالب $< 0$)، وإلا \textenglish{false}.
            \item في الفئة \textenglish{HibernationShip} يُعاد تعريف العملية (\textenglish{override})، وتعيد \textenglish{true} إذا كان \textbf{خزان المركبة فارغاً} (الوقود $\le 0$)، وإلا \textenglish{false}.
            \item في الفئة \textenglish{MotherShip} يُعاد تعريف العملية (\textenglish{override})، وتعيد \textenglish{true} إذا كانت السفينة الأم تستخدم من احتياطيها الطارئ \textbf{90\% أو أكثر} من الحجم المسموح لها (أي $\text{emergencyReserve} > 0$ و $\text{fuel} \le -0.9 \cdot \text{emergencyReserve}$)، وإلا \textenglish{false}.
        \end{itemize}
        \item أضيفوا إلى الفئة \textenglish{FleetControl} عملية باسم \textenglish{DangerShips()} تعيد مصفوفة (\textenglish{BasicShip[]}) تضم جميع المركبات التي \textbf{في خطر} (حسب نتيجة العملية \textenglish{IsInDanger()}).
    \end{enumerate}
    \textbf{الفحص في \textenglish{Main}:} \\
    استدعوا \textenglish{DangerShips()} في \textenglish{Main} واطبعوا جميع المركبات الموجودة في دائرة الخطر.

    \item \textbf{أذونات الوصول \textenglish{protected} والكلمة \textenglish{base} وسلسلة \textenglish{ToString}}
    \begin{itemize}
        \item أضيفوا إلى الفئة \textenglish{BasicShip} صفة \textbf{محمية} (\textenglish{protected}) من النمط \textenglish{string} باسم \textenglish{shipType} (نمط المركبة). تُحدَّد قيمتها من داخل كل فئة وارثة (\textenglish{"Scout"} أو \textenglish{"Hibernation"} أو \textenglish{"Mother"}).
        \item غيّروا العملية \textenglish{ToString} في \textenglish{BasicShip} بحيث تعيد بالضبط:
        \begin{center}
        \textenglish{\$"Type:\{shipType\}, Num:\{shipNumber\}, Captain:\{captainID\}, Fuel:\{fuel\}"}
        \end{center}
        \item في كل فئة وارثة، \textbf{أعيدوا تعريف} \textenglish{ToString} بحيث تستدعي \textenglish{base.ToString()} وتضيف إلى النتيجة الصفات الخاصة بالفئة (الاحتياطي الطارئ، تاريخ الوصول، اسم الأسطول).
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    اطبعوا مركز التحكم وتأكدوا من ظهور الوصف النصي الجديد لكل مركبة.

\end{enumerate}

\begin{boxAttention}[- نقطة للتفكير]
لماذا من الأفضل أن نستدعي \textenglish{base.ToString()} بدل أن نكتب الوصف من الصفر في كل فئة؟
\end{boxAttention}

\begin{enumerate}[resume, label=\textbf{تمرين \arabic*:}, wide=0pt, leftmargin=*, itemsep=2.2em]

    \item \textbf{عمليات خاصة بفئة فرعية واحدة فقط}
    \begin{itemize}
        \item أضيفوا إلى الفئة \textenglish{HibernationShip} عملية \textenglish{ZeroFuel()} تعيد \textenglish{true} إذا كان خزان المركبة فارغاً ($\text{fuel} \le 0$)، واستخدموها داخل \textenglish{IsInDanger()}.
        \item أضيفوا إلى الفئة \textenglish{ScoutShip} عملية \textenglish{LoanSuggestion()} تعيد سلسلة نصية: إذا كانت المركبة تستخدم الاحتياطي الطارئ ($\text{fuel} < 0$)، تعيد اقتراحاً بطلب \textbf{قرض وقود} من سفينة أم مع ذكر الكمية المقترحة (\textenglish{-fuel})، وإلا تعيد النص \textenglish{"No loan needed"}.
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    جرّبوا استدعاء هاتين العمليتين على كائنات من نمطيهما. ثم جرّبوا استدعاءهما من خلال متغير من النمط \textenglish{BasicShip} يشير إلى هذه الكائنات (وسجّلوا ملاحظتكم في تعليق).

    \item \textbf{العامل \textenglish{is} والتحويل لأسفل} \\
    أضيفوا إلى الفئة \textenglish{FleetControl} عملية \textenglish{PrintSuggestions()}. تمسح العملية مصفوفة المركبات (مع الانتباه إلى الحدود الفارغة \textenglish{null}) وتفعل ما يلي باستخدام العامل \textenglish{is} والتحويل لأسفل:
    \begin{itemize}
        \item إذا كانت المركبة من النمط \textenglish{HibernationShip} \textbf{وكان خزانها فارغاً} (\textenglish{ZeroFuel()}) — تطبع تحذيراً برقم المركبة.
        \item إذا كانت المركبة من النمط \textenglish{ScoutShip} أو \textenglish{MotherShip} — تطبع الاقتراح الناتج عن \textenglish{LoanSuggestion()} (أضيفوا عملية \textenglish{LoanSuggestion()} للفئة \textenglish{MotherShip} بصياغة مناسبة).
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    استدعوا \textenglish{PrintSuggestions()} في \textenglish{Main} وتأكدوا من طباعة التحذيرات والاقتراحات الصحيحة.

    \item \textbf{العامل \textenglish{as} ومقارنة الأساليب} \\
    أعيدوا كتابة العملية \textenglish{PrintSuggestions} في نسخة جديدة باسم \textenglish{PrintSuggestions2()} باستخدام العامل \textenglish{as} بدل \textenglish{is}. ثم أجيبوا في تعليق داخل الكود:
    \begin{enumerate}[label=\textbf{\arabic*.}]
        \item ما الفرق بين \textenglish{is} و \textenglish{as} وبين التحويل المباشر \textenglish{(HibernationShip)ship}؟
        \item ماذا يحدث في كل أسلوب إذا كانت المركبة من نمط آخر؟ وماذا إذا كانت \textenglish{null}؟
    \end{enumerate}
    \textbf{الفحص في \textenglish{Main}:} \\
    شغّلوا \textenglish{PrintSuggestions()} و \textenglish{PrintSuggestions2()} على نفس الأسطول وتأكدوا من تطابق المخرجات.

    \item \textbf{التعبئة الجماعية (\textenglish{RefuelAll}) — تعدد الأشكال في العمل} \\
    أضيفوا إلى الفئة \textenglish{FleetControl} عملية \textenglish{RefuelAll(double amount)} تتلقى كمية وقود وتُعبّئ بها \textbf{كل} المركبات في الأسطول، وتعيد عدد المركبات التي نجحت تعبئتها. \\
    \textbf{الفحص في \textenglish{Main}:} \\
    استدعوا \textenglish{RefuelAll} بكمية موجبة ثم بكمية سالبة واطبعوا عدد المركبات التي تمت تعبئتها وحالة الأسطول.

    \item \textbf{الوراثة من الفئة \textenglish{Object} — العملية \textenglish{Equals}}
    \begin{itemize}
        \item أعيدوا تعريف العملية \textenglish{Equals(object obj)} في الفئة \textenglish{BasicShip} بحيث تُعتبر المركبتان \textbf{متساويتين} إذا تطابقت لديهما قيم: رقم الأسطول، ورقم المحطة، ورقم المركبة (بغضّ النظر عن نمط المركبة أو كمية وقودها). انتبهوا لحالة \textenglish{null} ولحالة أن يكون المُعطى ليس من النمط \textenglish{BasicShip}.
        \item عدّلوا العملية \textenglish{Add} في \textenglish{FleetControl} بحيث \textbf{لا تضيف} مركبة موجودة مسبقاً في الأسطول (وتعيد \textenglish{false} في هذه الحالة).
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    اختبروا \textenglish{Equals} بين مركبتين بنفس الأرقام، وحاولوا إضافة مركبة مكررة إلى \textenglish{FleetControl} وتأكدوا من رفضها.

\end{enumerate}
