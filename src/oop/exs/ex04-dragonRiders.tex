\begin{boxDef}[قصة السلسلة: أكاديمية فرسان التنانين]
في وادٍ خفيّ بين الجبال تقع \textbf{أكاديمية فرسان التنانين}، حيث تعيش مئات التنانين ويتدرّب عليها الفرسان. أنتم المبرمجون المسؤولون عن \textbf{نظام تتبّع التنانين}: لكل تنين طاقة نار، وكهف يسكنه، وفارس مسؤول عنه. \\
ستبنون النظام خطوة بعد خطوة، وفي الطريق ستراجعون المصفوفات والشروط المركّبة والعمليات المساعدة، وستكتشفون لماذا ابتكر المبرمجون \textbf{الوراثة} و\textbf{تعدد الأشكال} و\textbf{الفئات المجردة} و\textbf{الواجهات}.
\end{boxDef}

\begin{boxWarning}
\textbf{قواعد ثابتة للعمل في جميع التمارين:}
\begin{itemize}
    \item في نهاية كل تمرين توجد فقرة \textbf{الفحص في \textenglish{Main}} تحدّد ما عليكم كتابته في العملية الرئيسية \textenglish{Main}: إنشاء كائنات، استدعاء العمليات الجديدة، تعديل الصفات، وطباعة الحالة بعد كل خطوة للتأكد من صحة النتائج. لا تحذفوا كود \textenglish{Main} القديم، بل حدّثوه ليواكب التغييرات الجديدة.
    \item \textbf{قاعدتان للكود النظيف:} (1) إذا كتبتم نفس المقطع مرتين — حوّلوه إلى \textbf{عملية مساعدة} (\textenglish{private}). (2) فكّروا في كل شرط: هل يمكن كتابته بشكل أوضح باستخدام \textenglish{\&\&} و \textenglish{||} و \textenglish{!}؟
\end{itemize}
\end{boxWarning}

\begin{enumerate}[label=\textbf{تمرين \arabic*:}, wide=0pt, leftmargin=*, itemsep=2.2em]

    \item \textbf{خريطة الوادي — الفئة \textenglish{Coordinates}} \\
    ابنوا فئة باسم \textenglish{Coordinates} تمثّل نقطة على خريطة الوادي. الخريطة مربّعة، وقيم الإحداثيات الصالحة هي من $0$ إلى $1000$ (شاملة):

    {\centering
    \renewcommand{\arraystretch}{1.2}
    \small
    \begin{tabular}{|>{\raggedleft\arraybackslash}p{0.4\textwidth}|>{\raggedright\arraybackslash}p{0.5\textwidth}|}
    \hline
    \multicolumn{2}{|c|}{\textbf{الفئة: \textenglish{Coordinates}}} \\
    \hline
    صفات (\textenglish{private}): الإحداثي الأفقي والإحداثي العمودي. & \textenglish{private int x, y;} \\
    \hline
    عملية بنائية تتلقى \textenglish{x} و \textenglish{y}، وعملية بنائية ناسخة تتلقى كائناً آخر من النمط \textenglish{Coordinates} وتنسخ قيمه. & \textenglish{public Coordinates(int x, int y)} \newline \textenglish{public Coordinates(Coordinates other)} \\
    \hline
    عمليات \textenglish{Get} و \textenglish{Set} لكل صفة، بحيث إذا كانت القيمة المُتلقّاة خارج المجال $0$–$1000$ تُضبط على أقرب حدّ (أقل من $0 \leftarrow 0$، وأكبر من $1000 \leftarrow 1000$). & \textenglish{GetX(), GetY(), SetX(int x), SetY(int y)} \\
    \hline
    عملية مساعدة واحدة تعيد القيمة بعد التصحيح، وتُستخدم في كل مكان يحتاج التصحيح (العملية البنائية وعمليتا \textenglish{Set}) بدل تكرار الشروط ثلاث مرات. & \textenglish{private int Clamp(int val)} \\
    \hline
    عملية تعيد النص بالشكل \textenglish{(x,y)}. & \textenglish{ToString()} \\
    \hline
    عملية تتلقى نقطة أخرى وتعيد المسافة بين النقطتين (\textenglish{double}). تذكّروا \textenglish{Math.Sqrt}. & \textenglish{DistanceTo(Coordinates other)} \\
    \hline
    عملية تعيد \textenglish{true} إذا كانت النقطة على \textbf{حافة} الخريطة (أي أن \textenglish{x} أو \textenglish{y} يساوي $0$ أو $1000$). & \textenglish{IsOnMapEdge()} \\
    \hline
    عملية تتلقى \textbf{زاويتين متقابلتين} لمستطيل وتعيد \textenglish{true} إذا كانت النقطة الحالية داخل المستطيل أو على حدوده. انتبهوا: الزاويتان قد تصلان \textbf{بأي ترتيب}، ولا يجوز أن تعتمد العملية على ترتيب معيّن. & \textenglish{IsInside(Coordinates c1, Coordinates c2)} \\
    \hline
    \end{tabular}
    \par} %

    \vspace{2em}
    \noindent \textbf{الفحص في \textenglish{Main}:}
    \begin{itemize}
        \item أنشئوا 4 نقاط، منها نقطة على الحافة ونقطة خارج المجال (مثلاً \textenglish{(1500, -20)}) وتأكدوا أنها صُحّحت، وغيّروا \textenglish{x} لنقطة بواسطة \textenglish{SetX} إلى قيمة خارج المجال واطبعوا النتيجة.
        \item احسبوا المسافة بين نقطتين (مثلاً \textenglish{(0,0)} و \textenglish{(3,4)} $\leftarrow 5$).
        \item جرّبوا \textenglish{IsInside} في ثلاث حالات (نقطة داخل، نقطة خارج، نقطة على الحدود)، وكرّروا الفحوصات بعد عكس ترتيب الزاويتين، واختبروا \textenglish{IsOnMapEdge}.
    \end{itemize}

    \item \textbf{التنين الناري — الفئة \textenglish{FireDragon}} \\
    ابنوا فئة باسم \textenglish{FireDragon} لإدارة تنين ناري "صيّاد" يخرج للصيد ويعود باستمرار:

    \hline
    عملية استهلاك طاقة إذا كان مسموحاً. الكمية يجب أن تكون موجبة، والاستهلاك مسموح فقط إذا لم تهبط الطاقة تحت \textenglish{-emberReserve}. تعيد \textenglish{true} إذا نجحت و \textenglish{false} إذا لم تنجح. & \textenglish{Consume(double amount)} \\

    \noindent \textbf{الفحص في \textenglish{Main}:}
    \begin{itemize}

        \item عبّئوا الأول بـ $100$ وحاولوا استهلاك $30$، ثم $80$ (تفشل)، ثم $-5$ (تفشل)، وللتنين الثاني جرّبوا استهلاك $40$ ثم $20$.
        \item اختبروا \textenglish{SetEmberReserve} و \textenglish{SetLair} و \textenglish{FlyTo}، وعدّلوا كائن \textenglish{Coordinates} الأصلي بعد إنشاء التنين للتأكد من أن عرين التنين لم يتأثر بفضل العملية البنائية الناسخة.
    \end{itemize}

    \textbf{الفحص في \textenglish{Main}:} \\
    أنشئوا كائناً من \textenglish{BasicDragon}، عبّئوه واطبعوه، وغيّروا موقع عرينه بواسطة \textenglish{SetLair} وتأكدوا من التغيير.

\begin{boxAttention}[- نقطة للتفكير]
هل كان أسهل لو طوّرنا الجزء المشترك \textbf{مرة واحدة} ثم أضفنا لكل فئة جزءها الخاص؟ وما الذي يجب أن \textbf{يبقى} في كل فئة فرعية ولا ينتقل إلى \textenglish{BasicDragon}؟ \\
\textbf{والآن ننتقل إلى تعلم الوراثة...}
\end{boxAttention}

\section*{المرحلة الثانية: تطبيق الوراثة بين الفئات}
% ==============================================================

\begin{enumerate}[resume, label=\textbf{تمرين \arabic*:}, wide=0pt, leftmargin=*, itemsep=2.2em]

    \item \textbf{تطبيق الوراثة في \textenglish{BasicDragon} و \textenglish{FireDragon} و \textenglish{HibernatingDragon}} \\
    حدّثوا الفئة \textenglish{BasicDragon} حسب قواعد البرمجة كائنية التوجه (صفات \textenglish{protected}، تغليف، عمليات بنائية)، بحيث تصبح \textbf{فئة أساسية (فئة أب)} لكل من \textenglish{FireDragon} و \textenglish{HibernatingDragon}. \\
    أعيدوا بناء الفئتين \textenglish{FireDragon} و \textenglish{HibernatingDragon} بحيث \textbf{ترثان} من \textenglish{BasicDragon} (استخدموا \textenglish{: base(...)} في العمليات البنائية)، مع الاحتفاظ بسجلّ الرحلات في \textenglish{FireDragon} وشرط الاستيقاظ في \textenglish{HibernatingDragon} دون إعادة كتابة ما ورثتاه. \\
    \textbf{الفحص في \textenglish{Main}:} \\
    شغّلوا كود الفحص في \textenglish{Main} وتأكدوا أن جميع النتائج بقيت صحيحة، واستدعوا عمليات موروثة وعمليات خاصة على كل كائن.

    \item \textbf{قائد القطيع — الفئة \textenglish{AlphaDragon} (وفيها مصفوفة تنانين)} \\
    أضيفوا فئة جديدة باسم \textenglish{AlphaDragon} تمثّل قائد قطيع من التنانين النارية وترث من \textenglish{BasicDragon}:

    {\centering
    \renewcommand{\arraystretch}{1.2}
    \small
    \begin{tabular}{|>{\raggedleft\arraybackslash}p{0.4\textwidth}|>{\raggedright\arraybackslash}p{0.5\textwidth}|}
    \hline
    \multicolumn{2}{|c|}{\textbf{الفئة الفرعية: \textenglish{AlphaDragon : BasicDragon}}} \\
    \hline
    صفات خاصة: اسم القطيع، جمر الطوارئ المسموح، ومصفوفة تحوي التنانين الأعضاء في القطيع (حجمها الأقصى يُحدَّد عند الإنشاء)، مع عدّاد لعدد الأعضاء الفعليين. & \textenglish{private string clanName;} \newline \textenglish{private double emberReserve;} \newline \textenglish{private FireDragon[] pack;} \newline \textenglish{private int packCount;} \\

    عملية بنائية أولى تتلقى بيانات المملكة والكهف والتنين، وهوية الفارس، والعرين، واسم القطيع، والحجم الأقصى للقطيع. & \textenglish{public AlphaDragon(int realmNumber, int caveNumber, int dragonNumber, string riderID, Coordinates lair, string clanName, int maxPackSize)} \\

    العمليتان \textenglish{Refuel} و \textenglish{Consume} كما في \textenglish{FireDragon}. & \textenglish{Refuel(double amount), Consume(double amount)} \\
    \hline
    عملية مساعدة تتلقى رقم تنين وتعيد مؤشره (مكانه) في المصفوفة، أو $-1$ إذا لم يوجد. & \textenglish{private int FindMemberIndex(int dragonNum)} \\
    \hline
    عملية تتلقى تنيناً نارياً وتضيفه في أول مكان فارغ. تنجح (\textenglish{true}) فقط إذا كان التنين غير \textenglish{null} وكان في القطيع متسع وكان غير موجود فيه مسبقاً (بحسب رقم التنين). & \textenglish{AddMember(FireDragon d)} \\
    \hline
    عملية تتلقى رقم تنين وتعيد الكائن نفسه إن وُجد في القطيع، وإلا \textenglish{null}. & \textenglish{FindMember(int dragonNum)} \\
    \hline
    عملية تتلقى رقم تنين وتحذفه من القطيع، ثم \textbf{تُزيح} كل الحدود التالية مكاناً واحداً إلى اليسار حتى لا تبقى ثغرة في وسط المصفوفة، وتعيد \textenglish{true} إذا حُذف. & \textenglish{RemoveMember(int dragonNum)} \\
    \hline
    عملية تتلقى مقداراً وكائناً من النمط \textenglish{FireDragon}، وتنقل الطاقة من القائد إلى التنين المُتلقّى. تنجح فقط إذا كان التنين غير \textenglish{null} وهو عضو فعلي في القطيع والمقدار موجب واستهلاك القائد لهذا المقدار مسموح. & \textenglish{TransferEnergy(double amount, FireDragon target)} \\

    \end{tabular}

    \noindent \textbf{الفحص في \textenglish{Main}:} \\
    أنشئوا قائداً بقطيع أقصى $3$ وأربعة تنانين نارية، واختبروا \textenglish{AddMember}، \textenglish{TransferEnergy}، و \textenglish{RemoveMember} في الحالات المختلفة.

\end{enumerate}

\section*{المرحلة الثالثة: سجلّ الوادي وتعدد الأشكال}

\begin{enumerate}[resume, label=\textbf{تمرين \arabic*:}, wide=0pt, leftmargin=*, itemsep=2.2em]

    \item \textbf{فئة سجلّ الوادي (\textenglish{DragonRegistry})} \\
    ابنوا فئة خدمات باسم \textenglish{DragonRegistry} تدير كل تنانين الوادي:

    {\centering
    \renewcommand{\arraystretch}{1.2}
    \small
    \begin{tabular}{|>{\raggedleft\arraybackslash}p{0.4\textwidth}|>{\raggedright\arraybackslash}p{0.5\textwidth}|}
    \hline
    \multicolumn{2}{|c|}{\textbf{الفئة: \textenglish{DragonRegistry}}} \\
    \hline
    صفات: مصفوفة تحوي تنانين من جميع الأنماط (\textenglish{FireDragon}، \textenglish{HibernatingDragon}، \textenglish{AlphaDragon})، وعدّاد لعدد التنانين الفعليين. & \textenglish{private BasicDragon[] dragons;} \newline \textenglish{private int count;} \\

    \hline
    عملية تضيف التنين في أول مكان فارغ، وتعيد \textenglish{true} إذا أُضيف و \textenglish{false} إذا كانت المصفوفة ممتلئة أو كان المُعطى \textenglish{null}. & \textenglish{Add(BasicDragon d)} \\

    عملية تحذف التنين وتُزيح الحدود التالية إلى اليسار (بلا ثغرات في الوسط) وتعيد \textenglish{bool}. & \textenglish{Remove(int dragonNum)} \\
    \hline
    عملية تعيد عدد التنانين الموجودين فعلياً، وعملية تعيد نصاً يصف كل التنانين في السجل. & \textenglish{Count(), ToString()} \\
    \hline
    \end{tabular}
    \par} %

    \vspace{2em}
    \noindent \textbf{الفحص في \textenglish{Main}:} \\
    أنشئوا سجلاً بحجم $6$، وأضيفوا إليه $5$ تنانين من الأنماط الثلاثة، وجرّبوا \textenglish{Find} و \textenglish{Remove} و \textenglish{Count} وطباعة السجل.

    \item \textbf{العمليات القابلة لإعادة التعريف (\textenglish{virtual} و \textenglish{override}): \textenglish{IsInDanger} و \textenglish{DangerLevel}}
    \begin{enumerate}[label=\textbf{(\arabic*)}]
        \item أضيفوا إلى المشروع عملية عامة قابلة لإعادة التعريف تعريفها \textenglish{public virtual bool IsInDanger()}:
        \begin{itemize}
            \item في \textenglish{BasicDragon}: تعيد \textenglish{false} دائماً.
            \item في \textenglish{FireDragon}: تعيد \textenglish{true} إذا كان التنين يستخدم جمر الطوارئ ($\text{energy} < 0$).
            \item في \textenglish{HibernatingDragon}: تعيد \textenglish{true} إذا كان التنين بلا طاقة ($\text{energy} \le 0$).
            \item في \textenglish{AlphaDragon}: تعيد \textenglish{true} إذا كان $\text{emberReserve} > 0$ واستخدم القائد $90\%$ أو أكثر من جمره ($\text{energy} \le -0.9 \cdot \text{emberReserve}$).
        \end{itemize}
        \item أضيفوا إلى \textenglish{DragonRegistry} عملية \textenglish{DangerDragons()} تعيد مصفوفة (\textenglish{BasicDragon[]}) بالتنانين التي في خطر.
        \item أضيفوا عملية قابلة لإعادة التعريف \textenglish{public virtual string DangerLevel()} تعيد \textenglish{"Safe"} أو \textenglish{"Warning"} أو \textenglish{"Critical"}:
        \begin{itemize}
            \item في \textenglish{BasicDragon}: دائماً \textenglish{"Safe"}.
            \item في \textenglish{FireDragon}: إذا لم تكن الطاقة سالبة $\leftarrow$ \textenglish{"Safe"}. وإذا كانت سالبة: إن استُهلك نصف جمر الطوارئ أو أكثر ($\text{energy} \le -0.5 \cdot \text{emberReserve}$) $\leftarrow$ \textenglish{"Critical"}، وإلا $\leftarrow$ \textenglish{"Warning"}.
            \item في \textenglish{HibernatingDragon}: \textenglish{"Critical"} إذا كانت الطاقة $0$ أو أقل، وإلا \textenglish{"Safe"}.
            \item في \textenglish{AlphaDragon}: إذا كان القائد في خطر (\textenglish{IsInDanger()}) $\leftarrow$ \textenglish{"Critical"}. وإلا إذا كانت طاقته سالبة $\leftarrow$ \textenglish{"Warning"}. وإلا افحصوا أعضاء القطيع: إذا كان أحدهم \textenglish{"Critical"} $\leftarrow$ \textenglish{"Warning"}، وإلا $\leftarrow$ \textenglish{"Safe"}.
        \end{itemize}
    \end{enumerate}
    \textbf{الفحص في \textenglish{Main}:} \\
    اطبعوا \textenglish{DangerLevel()} لكل تنين في السجل، ثم استدعوا \textenglish{DangerDragons()} واطبعوا النتيجة.

    \item \textbf{أذونات الوصول \textenglish{protected} والكلمة \textenglish{base} وسلسلة \textenglish{ToString}}
    \begin{itemize}
        \item أضيفوا إلى \textenglish{BasicDragon} صفة محمية \textenglish{protected string dragonType} تُحدَّد قيمتها في كل فئة وارثة (\textenglish{"Fire"} أو \textenglish{"Hibernating"} أو \textenglish{"Alpha"}).
        \item غيّروا \textenglish{ToString()} في \textenglish{BasicDragon} لتعيد:
        \begin{center}
        \textenglish{\$"Type:\{dragonType\}, Num:\{dragonNumber\}, Rider:\{riderID\}, Energy:\{energy\}"}
        \end{center}
        \item في كل فئة وارثة، أعيدوا تعريف \textenglish{ToString()} بحيث تستدعي \textenglish{base.ToString()} وتضيف الصفات الخاصة بها.
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    اطبعوا السجل وتأكدوا من صيغة \textenglish{ToString()} الجديدة لجميع التنانين.

\clearpage

    \item \textbf{عمليات خاصة بفئة فرعية واحدة فقط}
    \begin{itemize}
        \item أضيفوا إلى \textenglish{HibernatingDragon} عملية \textenglish{IsEmpty()} تعيد \textenglish{true} إذا كان التنين بلا طاقة ($\text{energy} \le 0$)، واستخدموها داخل \textenglish{IsInDanger()}.
        \item أضيفوا إلى \textenglish{FireDragon} عملية \textenglish{RescueSuggestion()} تعيد سلسلة نصية: إذا كانت طاقة التنين سالبة تقترح طلب تحويل طاقة من قائد قطيع بمقدار الدَّين مقرّباً لأعلى (\textenglish{Math.Ceiling(-energy)})، وإلا تعيد \textenglish{"No help needed"}.
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    استدعوا العمليتين على كائنات من نمطيهما في \textenglish{Main}.

    \item \textbf{العامل \textenglish{is} والتحويل لأسفل (\textenglish{PrintSuggestions})} \\
    أضيفوا إلى \textenglish{DragonRegistry} عملية \textenglish{PrintSuggestions()} تمسح كل تنانين السجل وتفعل ما يلي:
    \begin{itemize}
        \item إذا كان التنين \textenglish{HibernatingDragon} وكان فارغاً (\textenglish{IsEmpty()}) $\leftarrow$ تطبع تحذيراً برقم التنين.
        \item إذا كان \textenglish{FireDragon} وكان في خطر (\textenglish{IsInDanger()}) $\leftarrow$ تطبع \textenglish{RescueSuggestion()}.
        \item إذا كان \textenglish{AlphaDragon}: أضيفوا له عملية \textenglish{RescueSuggestion()}، واطبعوها فقط إذا كان القائد في خطر \textbf{أو} كان أحد أعضاء قطيعه في مستوى \textenglish{"Critical"}.
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    استدعوا \textenglish{PrintSuggestions()} على السجل وتأكدوا من طباعة الرسائل المتوقعة.

    \item \textbf{العامل \textenglish{as} ومقارنة الأساليب (\textenglish{PrintSuggestions2})} \\
    أعيدوا كتابة \textenglish{PrintSuggestions} في نسخة جديدة باسم \textenglish{PrintSuggestions2()} باستخدام العامل \textenglish{as} بدل \textenglish{is}، وقارنوا في تعليق بين \textenglish{is} و \textenglish{as} والتحويل المباشر. \\
    \textbf{الفحص في \textenglish{Main}:} \\
    شغّلوا \textenglish{PrintSuggestions()} و \textenglish{PrintSuggestions2()} وتأكدوا من تطابق المخرجات.

    \item \textbf{التغذية الجماعية (\textenglish{FeedAll} و \textenglish{TotalEnergy})} \\
    أضيفوا إلى \textenglish{DragonRegistry}:
    \begin{itemize}
        \item العملية \textenglish{FeedAll(double amount)}: تتلقى مقدار طاقة وتُعبّئ به كل تنانين السجل، وتعيد عدد التنانين التي نجحت تعبئتها.
        \item العملية \textenglish{TotalEnergy()}: تعيد مجموع طاقة كل التنانين في السجل.
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    اطبعوا \textenglish{TotalEnergy()} قبل وبعد استدعاء \textenglish{FeedAll(25)} و \textenglish{FeedAll(-10)}.

    \item \textbf{الوراثة من \textenglish{Object} — العملية \textenglish{Equals}}
    \begin{itemize}
        \item أعيدوا تعريف العملية \textenglish{Equals(object obj)} في \textenglish{BasicDragon} بحيث يعتبر التنينان متساويين إذا تطابقت لديهما قيم: رقم المملكة ورقم الكهف ورقم التنين.
        \item عدّلوا \textenglish{Add} في \textenglish{DragonRegistry} بحيث لا تضيف تنيناً موجوداً مسبقاً في السجل (بحسب \textenglish{Equals}).
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    اختبروا \textenglish{Equals} وحاولوا إضافة تنين مكرر إلى السجل.

    \item \textbf{دمج سجلّين (\textenglish{Merge})} \\
    أضيفوا إلى \textenglish{DragonRegistry} عملية \textenglish{Merge(DragonRegistry other)} تتلقى سجلاً آخر وتعيد \textbf{سجلاً جديداً} يحوي تنانين السجلّين معاً \textbf{دون تكرار} (باستخدام \textenglish{Add}). \\
    \textbf{الفحص في \textenglish{Main}:} \\
    أنشئوا سجلّين بينهما تنانين مشتركة، ادمجوهما عبر \textenglish{Merge}، واطبعوا السجل المدموج.

    \item \textbf{تحويل \textenglish{BasicDragon} إلى فئة مجردة (\textenglish{abstract})} \\
    غيّروا الفئة \textenglish{BasicDragon} بحيث تصبح \textbf{فئة مجردة} (\textenglish{abstract class})، وأضيفوا إليها عملية مجردة تعريفها:
    \begin{boxCode}
    \begin{english}
    \begin{minted}{csharp}
public abstract bool UseEnergy(object o);
    \end{minted}
    \end{english}
    \end{boxCode}
    وغيّروا \textenglish{IsInDanger()} لتصبح عملية مجردة (\textenglish{abstract}) أيضاً. طبّقوا \textenglish{UseEnergy(object o)} في الفئات الوارثة:
    \begin{itemize}
        \item \textenglish{FireDragon} و \textenglish{AlphaDragon}: تتلقى \textenglish{double} (أو \textenglish{int}) وتستهلكه كما في \textenglish{Consume}.
        \item \textenglish{HibernatingDragon}: تتلقى \textenglish{Date} (تاريخ اليوم) وتوقظ التنين كما في \textenglish{WakeUp(today, false)}.
        \item إذا وصل مُعطى من نمط غير مناسب (أو \textenglish{null}) تعيد \textenglish{false}.
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    امسحوا تنانين السجل واستدعوا \textenglish{UseEnergy(25.0)} و \textenglish{UseEnergy(new Date(...))} واطبعوا النتائج.

    \item \textbf{تطبيق الواجهة \textenglish{IComparable}} \\
    طبّقوا الواجهة \textenglish{IComparable} في الفئة \textenglish{Date}، وكذلك في الفئتين:
    \begin{itemize}
        \item \textbf{\textenglish{HibernatingDragon}:} تتم المقارنة حسب \textbf{تاريخ الاستيقاظ} (\textenglish{wakeDate}).
        \item \textbf{\textenglish{AlphaDragon}:} تتم المقارنة حسب \textbf{القوة الكامنة} ($\text{energy} + \text{emberReserve}$).
    \end{itemize}
    \textbf{الفحص في \textenglish{Main}:} \\
    قارنوا بين كائنات من كل فئة باستخدام \textenglish{CompareTo} واطبعوا النتائج.

        \item \textenglish{GetHibernatingByWakeDate()}: تعيد مصفوفة (\textenglish{HibernatingDragon[]}) من التنانين النائمة مرتّبة حسب \textbf{تاريخ الاستيقاظ} — الأقرب موعداً أولاً (تصاعدياً).

        \item أجروا إنقاذاً عبر \textenglish{TransferEnergy} من قائد القطيع إلى التنانين النارية التي في خطر، ثم أعلنوا إنذار بركان وأيقظوا التنانين النائمة عبر \textenglish{WakeUp(today, true)}.
        \item اطبعوا قادة القطعان مرتبين بالقوة الكامنة، والتنانين النائمة حسب موعد الاستيقاظ، ثم احذفوا التنانين التي استيقظت واطبعوا السجل النهائي.
    \end{enumerate}
