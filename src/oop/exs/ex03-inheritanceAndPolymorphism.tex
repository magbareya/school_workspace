\begin{boxDef}[مقدمة وسياق المسألة]
في شركة \textbf{"السعادة للجميع"}، يوجد 4 أنواع من الموظفين، بحيث ينتمي كل موظف إلى نوع واحد فقط:
\begin{itemize}
    \item \textbf{موظف عادي:} موظف عادي يعمل في الشركة.
    \item \textbf{عضو لجنة العمال:} موظف يشغل عضوية في نقابة/لجنة عمال الشركة.
    \item \textbf{تقني حواسيب:} موظف مسؤول عن صيانة عدد من الحواسيب (كل تقني مسؤول عن عدد محدد من الحواسيب).
    \item \textbf{مسؤول مشاريع:} موظف يدير مشاريع خاصة في الشركة، وهو مسؤول عن فريق مكوّن من 15 موظفاً يضم موظفين عاديين، وأعضاء لجنة، وتقنيين.
\end{itemize}
\end{boxDef}

\begin{boxNote}[شروط التوظيف والترقية وتحديد الرواتب]
\begin{itemize}
    \item \textbf{التوظيف والترقية:} عند التعيين الجديد يتم قبول موظفين عاديين، كما يمكن توظيف تقنيين مباشرة. وبموجب قرار مجلس إدارة الشركة، فإن أعضاء لجنة العمال ومسؤولي المشاريع تتم ترقيتهم حصراً من كادر الموظفين العاديين أو التقنيين.
    \item \textbf{سلّم الرواتب الشهري:} يتحدد راتب الموظف وفق الراتب الأساسي، والأقدمية، وعلاوات إضافية:
    \begin{itemize}
        \item الراتب الأساسي الابتدائي: $5000$ ش.ج.
        \item عن كل سنة أقدمية: علاوة قدرها $500$ ش.ج.
        \item عن كل 10 حواسيب (للتقني): علاوة قدرها $250$ ش.ج \\ (أي: $\lfloor \frac{\text{computers}}{10} \rfloor \cdot 250$).
        \item منحة إدارة خاصة (لمسؤول المشاريع): علاوة قدرها $100$ ش.ج عن كل تقني، وعلاوة قدرها $50$ ش.ج عن كل عضو لجنة عمال ضمن فريق مشروعه المكون من 15 موظفاً.
    \end{itemize}
\end{itemize}
\end{boxNote}

\section*{نظام التصويت ونقاط التأثير}
تُتخذ القرارات في الشركة عبر التصويت بمشاركة جميع الموظفين. يتم التصويت بواسطة \textbf{نقاط التأثير}، حيث يمتلك كل موظف عدداً مختلفاً من النقاط وفق نوعه وأقدميته كما هو موضح في الجدول التالي:

\begin{center}
\renewcommand{\arraystretch}{1.6}
{\small
\begin{tabular}{|p{0.22\textwidth}|p{0.72\textwidth}|}
\hline
\multicolumn{1}{|c|}{\textbf{نوع الموظف}} & \multicolumn{1}{c|}{\textbf{عدد نقاط التصويت}} \\
\hline
\textbf{موظف عادي} \newline (\textenglish{Employee}) &
يحصل على 4 نقاط لكونه موظفاً في الشركة، ونقطة إضافية عن كل سنة أقدمية في الشركة. \newline
\textbf{مثال:} موظف عادي لديه 5 سنوات أقدمية يحصل على $4 + 5 = 9$ نقاط. \\
\hline
\textbf{عضو لجنة عمال} \newline (\textenglish{UnionMember}) &
يحصل على \textbf{ضعف} عدد النقاط التي يستحقها الموظف العادي، بالإضافة إلى نقطتين عن كل سنة يشغل فيها عضوية لجنة العمال. \newline
\textbf{مثال:} عضو لجنة لديه 5 سنوات أقدمية في الشركة ومن بينها 3 سنوات في اللجنة، يحصل على: $2 \cdot 9 + 2 \cdot 3 = 18 + 6 = 24$ نقطة. \\
\hline
\textbf{تقني حواسيب} \newline (\textenglish{Technician}) &
يحصل على نقاط مثل الموظف العادي، بالإضافة إلى نقطة واحدة عن كل حاسوب تحت مسؤوليته. \newline
\textbf{مثال:} تقني لديه 5 سنوات أقدمية ومسؤول عن 4 حواسيب، يحصل على: $9 + 4 = 13$ نقطة. \\
\hline
\textbf{مسؤول مشاريع} \newline (\textenglish{Supervisor}) &
يحصل على نقاط مثل الموظف العادي، بالإضافة إلى مجموع نقاط جميع الموظفين الـ 15 الذين يشرف عليهم. \newline
\textbf{مثال:} مسؤول مشاريع لديه 5 سنوات أقدمية ومجموع نقاط الـ 15 موظفاً في فريقه هو 80 نقطة، يحصل على: $9 + 80 = 89$ نقطة. \\
\hline
\end{tabular}
}
\end{center}

ترغب إدارة الشركة في بناء نظام محوسب يحسب ويعرض تكلفة الراتب الشهري وعدد نقاط التصويت لكل موظف. لهذا الغرض، تقرر تعريف الفئات التالية:
\textenglish{Employee}، \textenglish{UnionMember}، \textenglish{Technician}، \textenglish{Supervisor}، وفئة الشركة \textenglish{Company}.

\section*{المهام والمطلوب}

حل البنود التالية بأفضل صورة تراعي مبادئ البرمجة كائنية التوجه:

\begin{enumerate}[label=\textbf{\alph*.} , itemsep=2.5em]

    \item \textbf{تطبيق فئات الموظفين والعمليات البنائية:} \\
    اكتب تطبيقاً كاملاً لتعريف وخصائص والعمليات البنائية لفئات الموظفين الأربع: \newline
    \textenglish{Employee}، \textenglish{Technician}، \textenglish{UnionMember}، و \textenglish{Supervisor}، \newline
    مع الاستفادة من مبدأ الوراثة واستدعاء العملية البنائية للفئة الأساسية (\textenglish{base}) كما هو موضح في الجداول أعلاه.

    \item \textbf{العملية \textenglish{ToString}:} \\
    طبّق العملية \textenglish{ToString} في كل فئة من فئات الموظفين الأربع، بحيث تعيد وصفاً نصياً شاملاً للموظف وقيمه، مع تجنب تكرار الكود عبر استدعاء \textenglish{base.ToString()}.

    \item \textbf{العمليتان \textenglish{GetScore} و \textenglish{GetSalary}:} \\
    طبّق العمليتين \textenglish{GetScore} و \textenglish{GetSalary} في كل فئة من فئات الموظفين الأربع، مع الاستفادة من تعدد الأشكال واستدعاء عمليات الفئة الأساسية لتجنب تكرار العمليات الحسابية.

    \item \textbf{عملية المكافأة السنوية (\textenglish{YearlyBonus}):} \\
    حققت الشركة أرباحاً ممتازة وترغب في توزيع مكافأة سنوية للموظفين. حددت الشركة استحقاق المكافأة للموظفين العاديين والتقنيين \textbf{فقط} وفق الشروط التالية:
    \begin{itemize}
        \item أقدمية 5 سنوات على الأقل في الشركة.
        \item بالنسبة للتقنيين: مسؤولية صيانة 100 حاسوب على الأقل بالإضافة لشرط الأقدمية.
    \end{itemize}
    طبّق العملية \textenglish{YearlyBonus} في الفئات المناسبة بحيث تفحص هل يستحق الموظف المكافأة السنوية، وبما يتوافق مع مبادئ الوراثة وتعدد الأشكال وهندسة البرمجيات.
\clearpage
    \item \textbf{تطبيق فئة الشركة (\textenglish{Company}):} \\
    اكتب تطبيقاً كاملاً لفئة الشركة \textenglish{Company} يشمل الثابت، الخصائص، العملية البنائية، وعمليتي \textenglish{AddEmployee} و \textenglish{TotalSalary} الموضحة في جدول الفئة أعلاه. وضّح علاقة هذه الفئة بهرمية الفئات.

    \item \textbf{عملية فحص قبول الاقتراح (\textenglish{IsAccepted}):} \\
    طُرح اقتراح رسمي في الشركة لتغيير اسمها من "السعادة للجميع" إلى \textbf{"السعادة الآن وفي المستقبل"}. \\
    صوّت جميع الموظفين العاديين والتقنيين \textbf{تأييداً} للاقتراح (مع)، بينما عارض جميع أعضاء لجنة العمال ومسؤولي المشاريع هذا الاقتراح (\textbf{ضد}). \\
    يُعتمد القرار فقط إذا كان مجموع نقاط المؤيدين \textbf{أكبر تماماً} من مجموع نقاط المعارضين. \\
    أمامك تعريف عملية عامة وثابتة:
\begin{boxCode}
\begin{english}
\begin{minted}{csharp}
public static bool IsAccepted(object[] arr)
\end{minted}
\end{english}
\end{boxCode}
    تتلقى العملية مصفوفة كائنات عامة تمثل موظفي الشركة، وتُعيد \textenglish{true} إذا قُبل الاقتراح، وإلا تُعيد \textenglish{false}. اكتب تطبيقاً كاملاً لهذه العملية.

\end{enumerate}

\begin{boxSolution}[الفروع هـ + و: فئة الشركة \textenglish{Company}]
العلاقة بين \textenglish{Company} و \textenglish{Employee} هي علاقة احتواء وتركيب وليست وراثة.
{\small
\begin{english}
\begin{minted}{csharp}
public class Company
{
    public const int MAX = 100;
    private Employee[] employees;
    private int current;

    public Company()
    {
        this.employees = new Employee[MAX];
        this.current = 0;
    }

    public bool AddEmployee(Employee e)
    {
        if (this.current < MAX && e != null)
        {
            this.employees[this.current] = e;
            this.current++;
            return true;
        }
        return false;
    }

    public double TotalSalary()
    {
        double total = 0;
        for (int i = 0; i < this.current; i++)
        {
            total += this.employees[i].GetSalary();
        }
        return total;
    }
}
\end{minted}
\end{english}
}
\end{boxSolution}

\begin{boxSolution}[الفرع ح: عملية التصويت العامة \textenglish{IsAccepted}]
المؤيدون هم الموظفون العاديون والتقنيون، بينما المعارضون هم أعضاء لجنة العمال ومسؤولو المشاريع:
{\small
\begin{english}
\begin{minted}{csharp}
public static bool IsAccepted(object[] arr)
{
    if (arr == null)
    {
        return false;
    }

    int forPoints = 0;
    int againstPoints = 0;

    for (int i = 0; i < arr.Length; i++)
    {
        if (arr[i] is Employee emp)
        {
            if (emp is UnionMember || emp is Supervisor)
            {
                againstPoints += emp.GetScore();
            }
            else // Regular Employee or Technician
            {
                forPoints += emp.GetScore();
            }
        }
    }

    return forPoints > againstPoints;
}
\end{minted}
\end{english}
}
