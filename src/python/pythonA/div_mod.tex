\section*{القسم الأول: أسئلة الدرس - القسمة الصحيحة وباقي القسمة}

\begin{enumerate}[label=\textbf{تمرين \arabic*:}, wide=0pt, itemsep=2.2em]

% ---------------------------------------------------------------
\item \textbf{توزيع هدايا:} نريد أن نوزّع هدايا عيد ميلاد لخمسة أولاد بالتساوي.
اكتبوا عملية تستقبل كمية الهدايا، تحسب وتطبع كمية الهدايا التي يحصل عليها كل ولد وكمية الهدايا المتبقية.

\ifdetailed
\begin{boxExample}
عند إدخال $34$ هدية مقسّمة على $5$ أولاد:
\begin{english}
\begin{minted}{text}
Enter number of presents: 34
Each kid gets: 6 presents
Left presents: 4 presents
\end{minted}
\end{english}
\end{boxExample}
\fi

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def presents():
    num = int(input("Enter number of presents: "))
    print("Each kid gets:", num // 5, "presents")
    print("Left presents:", num % 5, "presents")

presents()
\end{minted}
\end{english}
\end{boxSolution}
\fi
\filbreak

% ---------------------------------------------------------------
\item \textbf{قسمة عادية، قسمة صحيحة وباقي القسمة:} اكتبوا عملية تستقبل عددين صحيحين وتطبع نتيجة قسمتهما على بعض، نتيجة القسمة الصحيحة، والباقي من القسمة. يجب أن يظهر التمرين والحل على الشاشة.

\ifdetailed
\begin{boxExample}
عند إدخال $\text{num1} = 7$ و $\text{num2} = 5$:
\begin{english}
\begin{minted}{text}
Enter number: 7
Enter number: 5
num1 // num2 : 1
num1 / num2 : 1.4
num1 % num2 : 2
\end{minted}
\end{english}
\end{boxExample}
\fi

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def ex1():
    num1 = int(input("Enter number: "))
    num2 = int(input("Enter number: "))
    print("num1 // num2 :", num1 // num2)
    print("num1 / num2 :", num1 / num2)
    print("num1 % num2 :", num1 % num2)

ex1()
\end{minted}
\end{english}
\end{boxSolution}
\fi
\filbreak

% ---------------------------------------------------------------
\item \textbf{القسمة على $2$:} أمامكم عملية تطبع باقي القسمة على $2$ ($0$ أو $1$) مع عبارة توضيحية ملائمة:

\begin{boxCode}
\begin{english}
\begin{minted}{python}
def ex2():
    num = int(input("Enter an integer number:"))
    print("Calculating modulo:")
    print(num % 2)

ex2()
\end{minted}
\end{english}
\end{boxCode}

% ---------------------------------------------------------------
\item \textbf{منازل العدد:} اكتبوا عملية تستقبل عددًا ثنائي المنازل (مؤلّفًا من منزلتين)، وتطبع:
\begin{itemize}
\item منزلة الآحاد مع رسالة ملائمة.
\item منزلة العشرات مع رسالة ملائمة.
\item حاصل جمع المنزلتين مع رسالة ملائمة.
\end{itemize}

\ifwithsols
\begin{boxSolution}

\begin{minted}{python}

    units = num % 10
    tens = num // 10
    total = units + tens
    print("The units digit is:", units)
    print("The tens digit is:", tens)
    print("Sum of two digits:", total)

ex3()
\end{minted}
\end{english}
\end{boxSolution}

% ---------------------------------------------------------------
\item \textbf{عدد ثلاثي المنازل:} أمامكم عمليتان تستقبل كلٌّ منهما عددًا ثلاثي المنازل (مؤلّفًا من $3$ منازل) وتطبع حاصل ضرب منازله الثلاث:

\begin{boxCode}
\begin{english}
\begin{minted}{python}
def mult1():
    num = int(input("Enter 3 digits num: "))
    d = num % 10
    t = num // 10 % 10
    h = num // 100
    print(d * t * h)

mult1()
\end{minted}
\end{english}
\end{boxCode}

\begin{boxCode}
\begin{english}
\begin{minted}{python}
def mult2():
    num = int(input("Enter 3 digits num: "))
    d = num % 10
    t = num % 100 // 10
    h = num // 100
    print(d * t * h)

mult2()
\end{minted}
\end{english}
\end{boxCode}

\begin{enumerate}[label=\alph*.]
\item ما هو المُخرج الناتج لكلّ من العمليتين عند إدخال العدد $243$؟
\ifwithsols
\begin{boxSolution}
المُخرج في كلتا العمليتين هو $24$ (لأنّ $3 \cdot 4 \cdot 2 = 24$).
\end{boxSolution}
\fi
\filbreak

\item اشرحوا كيف يتمّ استخلاص منزلة العشرات $t$ في كلّ من العمليتين.
\ifwithsols
\begin{boxSolution}
\begin{itemize}
\item في \textenglish{mult1}: التعبير \textenglish{\texttt{num // 10}} يحذف منزلة الآحاد (مثلًا $243 // 10 = 24$)، ثم \textenglish{\texttt{\% 10}} يستخلص منزلة الآحاد للعدد الناتج وهي منزلة العشرات الأصلية ($24 \% 10 = 4$).
\item في \textenglish{mult2}: التعبير \textenglish{\texttt{num \% 100}} يستخلص آخر منزلتين (الآحاد والعشرات معًا، مثلًا $243 \% 100 = 43$)، ثم \textenglish{\texttt{// 10}} يستخلص منزلة العشرات ($43 // 10 = 4$).
\end{itemize}
\end{boxSolution}
\fi
\filbreak
\end{enumerate}
\filbreak

\item \textbf{باقات الزهور:} يتلقّى حانوت بيع الزهور يوميًا كميات مختلفة من الزهور. تقوم سهى بتنسيق الزهور في الحانوت وتقسيمها لباقات بالتساوي.
اكتبوا عملية لمساعدة سهى على تقسيم كميات الزهور اليومية لباقات تحتوي على عدد زهور متساوٍ: تستقبل العملية عدد الورود في المخزون اليومي الذي استلمه الحانوت وعدد الباقات التي يجب على سهى تنسيقها، وتطبع عدد الورود في كل باقة وعدد الورود المتبقية.

\ifdetailed
\begin{boxExample}
\begin{itemize}
\item لإدخال $1000$ وردة و $23$ باقة: في كل باقة $43$ وردة، ويتبقى $11$ وردة.
\item لإدخال $610$ وردات و $20$ باقة: في كل باقة $30$ وردة، ويتبقى $10$ وردات.
\end{itemize}
\end{boxExample}
\fi

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def ex4():
    flowers = int(input("Enter the number of flowers today: "))
    bouquet = int(input("Enter the number of ordered bouquets: "))
    print("The number of flowers to be assigned to each bouquet is:")
    print(flowers // bouquet)
    print("The number of flowers left is:")
    print(flowers % bouquet)

ex4()
\end{minted}
\end{english}
\end{boxSolution}
\fi
\filbreak

\item \textbf{كتب المكتبة:} بمناسبة افتتاح مكتبة المدرسة يقومون بترتيب الرفوف فيها. في كل رفّ يمكن وضع $15$ كتابًا تعليميًا، وتوضع الكتب التي لا مكان لها على الرفوف جانبًا في العربة.
اكتبوا عملية تستقبل عدد الكتب، ثم تحسب وتطبع عدد الرفوف اللازم لحفظها، وتطبع عدد الكتب المتبقية التي لا مكان لها على الرفوف.

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def ex5():
    books_num = int(input("Enter number of books: "))
    shelf = 15
    print("The number of shelves needed:", books_num // shelf)
    print("The number of books in the cart:", books_num % shelf)

ex5()
\end{minted}
\end{english}
\end{boxSolution}
\fi
\filbreak

% ---------------------------------------------------------------
\item \textbf{فعاليات مدرسية:} في المدرسة يخرجون ليوم فعاليات مدرسية لكل صفوف الطبقة. يوجد $3$ باصات، في كل باص $40$ مكانًا للجلوس، ويجب ملء الأماكن في الباصات بالكامل قبل الانطلاق.
\begin{itemize}
\item وصل من الصف السابع "أ": $23$ طالبًا، مرافقان اثنان ($2$) ومسعف ($1$).
\item وصل من الصف السابع "ب": $20$ طالبًا ومرافق واحد ($1$).
\item وصل من الصف السابع "ج": $30$ طالبًا و $3$ مرافقين.
\end{itemize}

\begin{enumerate}[label=\alph*.]
\item هل يوجد مكان لكل الطلاب والمرافقين (عددهم الكلي $80$) في الباصات الممتلئة، أم توجد حاجة لإحضار حافلة صغيرة (مينيبوس)؟
\ifwithsols
\begin{boxSolution}
بما أنّ كل باص يتّسع لـ $40$ مسافرًا وعدد المسافرين الكلي هو $80$، فإنّهم يملؤون باصين بالكامل ($80 // 40 = 2$) ولا يتبقى أي مسافر بدون مقعد ($80 \% 40 = 0$)، لذا لا توجد حاجة لإحضار مينيبوس.
\end{boxSolution}
\fi
\filbreak

\item اكتبوا عملية تستقبل عدد المسافرين وعدد الأماكن في الباص الواحد، وتطبع عدد المسافرين الذين لا يوجد لهم مكان في الباصات الممتلئة.
\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def ex6():
    people = int(input("Enter the number of people: "))
    seats = int(input("Enter the number of seats: "))
    left = people % seats
    print("The number of people with no seat:", left)

ex6()
\end{minted}
\end{english}
\end{boxSolution}
\fi
\filbreak
\end{enumerate}
\filbreak

\end{enumerate}

    print("Cookies cost:", cookies_cost, "Cakes cost:", cakes_cost)
    print("Total cost:", total, "nis")

bakery()
