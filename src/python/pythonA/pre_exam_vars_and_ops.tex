\item معطاة ثلاثة متغيرات \textenglish{a}، \textenglish{b}، \textenglish{c} وقيمها كالتالي:
\begin{center}
\textenglish{a = 10, b = 15, c = 0}
\end{center}
احسبوا قيمة كل من التعابير المنطقية في الجدول، واكتبوا القيمة (\textenglish{True} / \textenglish{False}) في العمود الفارغ.

{\centering
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{|C{1.2cm}|>{\raggedright\arraybackslash}p{9.6cm}|C{3cm}|}
\hline
\textbf{البند} & \centering\textbf{التعبير المنطقي} & \textbf{قيمة التعبير المنطقي} \\
\hline
$1$ & \textenglish{\texttt{a >= b or b == c}} & \ifwithsols \textenglish{False} \fi \\ \hline
$2$ & \textenglish{\texttt{a \% 5 == 0 and b // 5 > c}} & \ifwithsols \textenglish{True} \fi \\ \hline
$3$ & \textenglish{\texttt{a < 0 or (b == c + 15 and a >= c)}} & \ifwithsols \textenglish{True} \fi \\ \hline
$4$ & \textenglish{\texttt{a \% 2 == c \% 2 and b \% 3 > a \% 3}} & \ifwithsols \textenglish{False} \fi \\ \hline
$5$ & \textenglish{\texttt{a != b and b != c}} & \ifwithsols \textenglish{True} \fi \\ \hline
\end{tabular}
\par} %

\ifwithsols
\begin{boxSolution}
\begin{enumerate}[label=\arabic*.]
\item $10 \geq 15$ خطأ، و $15 = 0$ خطأ، إذن \textenglish{False}.
\item $10 \% 5 = 0$ صحيح، و $15 // 5 = 3 > 0$ صحيح، إذن \textenglish{True}.
\item $10 < 0$ خطأ، لكن $15 = 0 + 15$ صحيح و $10 \geq 0$ صحيح، إذن \textenglish{True}.
\item $10 \% 2 = 0$ و $0 \% 2 = 0$ متساويان (صحيح)، لكن $15 \% 3 = 0$ ليس أكبر من $10 \% 3 = 1$ (خطأ)، إذن \textenglish{False}.
\item $10 \neq 15$ صحيح، و $15 \neq 0$ صحيح، إذن \textenglish{True}.
\end{enumerate}
\end{boxSolution}
\fi
