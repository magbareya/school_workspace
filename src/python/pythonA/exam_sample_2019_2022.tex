\item معطاة ثلاثة متغيرات $a$، $b$، $c$ وقيمها كالتالي:
$$a = 10, \quad b = 15, \quad c = 0$$
احسبوا قيمة كل واحد من التعابير البوليانية في الجدول، واكتبوا القيمة (\textenglish{True} / \textenglish{False}) في العمود الفارغ.

{\renewcommand{\arraystretch}{1.4}
\begin{longtable}{|C{1.5cm}|>{\raggedright\arraybackslash}p{9cm}|C{3.5cm}|}
\hline
\textbf{البند} & \centering\textbf{تعبير بولياني} & \textbf{قيمة التعبير} \\
\hline
$1$. & \textenglish{\texttt{a >= b or b == c}} & \ifwithsols \textenglish{False} \fi \\ \hline
$2$. & \textenglish{\texttt{a \% 5 == 0 and b // 5 > c}} & \ifwithsols \textenglish{True} \fi \\ \hline
$3$. & \textenglish{\texttt{a < 0 or (b == c + 15 and a >= c)}} & \ifwithsols \textenglish{True} \fi \\ \hline
$4$. & \textenglish{\texttt{a \% 2 == c \% 2 and b \% 3 > a \% 3}} & \ifwithsols \textenglish{False} \fi \\ \hline
$5$. & \textenglish{\texttt{a != b and b != c}} & \ifwithsols \textenglish{True} \fi \\ \hline
\end{longtable}}

\ifwithsols
\begin{boxSolution}
\begin{enumerate}[label=\arabic*.]
\item $10 \geq 15$ خطأ، و $15 = 0$ خطأ، إذن \textenglish{False}.
\item $10 \% 5 = 0$ صحيح، و $15 // 5 = 3 > 0$ صحيح، إذن \textenglish{True}.
\item $10 < 0$ خطأ، لكن $15 = 0 + 15$ صحيح و $10 \geq 0$ صحيح، إذن \textenglish{True}.
\item $10 \% 2 = 0$ و $0 \% 2 = 0$ متساويان (صحيح)، لكن $15 \% 3 = 0$ ليس أكبر من $10 \% 3 = 1$ (خطأ)، إذن \textenglish{False}.
\item $10 \neq 15$ صحيح، و $15 \neq 0$ صحيح، إذن \textenglish{True}.
\end{enumerate}
\end{boxSolution}
\fi

\item أمامكم ثلاثة مقاطع برنامج. الهدف من كل واحد من مقاطع البرنامج هو استقبال $10$ أعداد صحيحة، وطباعة كمية الأعداد \textbf{ثنائية المنازل الموجبة} (من $10$ إلى $99$) التي ظهرت في المُدخل.

\textbf{مثال:} لسلسلة الأعداد العشرة التالية: $230$، $-555$، $76$، $5$، $80$، $-100$، $77$، $8$، $74$، $90$، يُطبع العدد $5$.
في كل مقطع برنامج وقع خطأ منطقي واحد، ولذلك لا يحقّق المقطع هدفه. اشرحوا الخطأ وصحّحوه.
(ملاحظة: الخطأ ليس خطأ وقت تشغيل. افترضوا أنّ المُدخل سليم ويشمل أعدادًا صحيحة فقط.)

\begin{enumerate}[label=\textbf{مقطع \alph*:}, wide=0pt, itemsep=1.5em]

\begin{boxCode}

count = 0
index = 0
num = int(input("Enter an integer number"))
while index < 10:
    if num <= 99 and num >= 10:
        count = count + 1
    index = index + 1
print("count", count)
\end{minted}
\end{english}
\end{boxCode}
\ifwithsols
\begin{boxSolution}
\textbf{الخطأ:} أمر الاستقبال \textenglish{\texttt{num = int(input(...))}} مكتوب خارج الحلقة فقط، ولذلك يستقبل البرنامج عددًا واحدًا فقط ويفحصه $10$ مرات بدل استقبال $10$ أعداد مختلفة. \\
\textbf{التصحيح:} نقل السطر \textenglish{\texttt{num = int(input("Enter an integer number"))}} إلى داخل حلقة \textenglish{while}.
\end{boxSolution}
\else
\vspace{0.5cm}

\textbf{كيف يجب تصحيحه؟} \hrulefill

\item متوالية فيبوناتشي (\textenglish{Fibonacci}) هي متوالية يكون حدّاها الأولان $1$، وكل حدّ يليهما يساوي مجموع الحدّين اللذين سبقاه.

\item \textbf{العدد المثالي} هو عدد يساوي مجموع جميع قواسمه الموجبة (لا يشمل العدد نفسه).
\begin{itemize}
\item \textbf{مثال:} العدد $28$ هو عدد مثالي لأنّه يساوي مجموع جميع قواسمه: $28 = 1 + 2 + 4 + 7 + 14$.
\item القاسم هو عدد يقسم العدد المقسوم مع باقي قسمة صفر. مثلًا: $4$ هو قاسم لـ $24$، لكنّه ليس قاسمًا لـ $23$.
\end{itemize}

\begin{enumerate}[label=\alph*.]
\item حسب التعريف أعلاه، هل $6$ هو عدد مثالي؟ إذا نعم، اعرضوا تعبيرًا رياضيًا (كما هو معروض في المثال) يُظهر ذلك؛ وإلّا اشرحوا.
\ifwithsols
\begin{boxSolution}
\textbf{نعم}، العدد $6$ هو عدد مثالي لأنّ قواسمه (بدون الـ $6$ نفسه) هي $1$، $2$، $3$، ومجموعها يساوي:
$$6 = 1 + 2 + 3$$
\end{boxSolution}
\else
\\ \underline{\hspace{14cm}}
\fi
\filbreak
