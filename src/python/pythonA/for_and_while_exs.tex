\subsection*{سؤال 2}
طُلب من التلاميذ أن يكتبوا برنامجًا يستقبل 5 أزواج أعداد ويطبع \textbf{عدد الأزواج} التي لهما نفس باقي القسمة على 2 (أي عددين زوجيين أو عددين فرديين).

\textbf{مثال:}
\begin{itemize}
    \item زوج الأعداد الأول: \textenglish{4 \ 8} $\leftarrow$ نفس باقي القسمة على 2، العددين زوجيين.
    \item زوج الأعداد الثاني: \textenglish{7 \ 66} $\leftarrow$ باقي القسمة على 2 مختلف.
    \item زوج الأعداد الثالث: \textenglish{3 \ 5} $\leftarrow$ نفس باقي القسمة على 2، العددين فرديين.
    \item زوج الأعداد الرابع: \textenglish{1 \ 8} $\leftarrow$ باقي القسمة على 2 مختلف.
    \item زوج الأعداد الخامس: \textenglish{1 \ 5} $\leftarrow$ نفس باقي القسمة على 2، العددين فرديين.
\end{itemize}
المخرج الناتج من البرنامج هو \textenglish{3}، أي عدد الأزواج ذات نفس باقي القسمة على 2 هي \textenglish{3}.

أكملوا مقطع الكود التالي بحيث يقوم بالمطلوب.

\begin{boxCode}
\begin{english}
\begin{minted}{python}
count = ______________
for item in range(______________):
    num1 = int(input("Please enter first number: "))
    num2 = int(input("Please enter second number: "))
    if ______________:
        count = ______________
print(______________)
\end{minted}
\end{english}
\end{boxCode}

\textbf{مخزن الكلمات:}
\begin{center}
\begin{english}
\texttt{count} \quad \texttt{count + 1} \quad \texttt{num1\%2 == num2\%2} \quad \texttt{5} \quad \texttt{0}
\end{english}
\end{center}

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
count = 0
for item in range(5):
    num1 = int(input("Please enter first number: "))
    num2 = int(input("Please enter second number: "))
    if num1 % 2 == num2 % 2:
        count = count + 1
print(count)
\end{minted}
\end{english}
\end{boxSolution}

\section*{القسم الثاني: حلقات \textenglish{while}}

الحلقة \textbf{اللانهائية} هي الحلقة التي شرط التوقف فيها لا يتحقق أبدًا، ولذلك الأوامر التي في جسم الحلقة تتنفذ إلى ما لا نهاية. أمامكم مقاطع برنامج تحتوي على حلقة \textenglish{while}. لكل مقطع برنامج، حدّدوا إذا كانت الحلقة لا نهائية أم نهائية.
إذا كانت الحلقة لا نهائية، أشيروا بـ \textenglish{X} في الخانة الملائمة، وإذا كانت نهائية اكتبوا ماذا سيكون المخرج (أو عدد المرات التي ستُنفّذ فيها الحلقة).

\ifwithsols
\begin{boxSolution}
\begin{enumerate}[label=\arabic*., itemsep=1em]
    \item حلقة \textbf{نهائية}: المخرج \textenglish{\texttt{5, 4, 3, 2, 1}} وتُنفَّذ الحلقة 5 مرات.
    \item حلقة \textbf{نهائية}: المخرج \textenglish{\texttt{0, 3, 6, 9}} وتُنفَّذ الحلقة 4 مرات.
    \item حلقة \textbf{نهائية}: الشرط غير محقق منذ البداية، لذلك لا يوجد مخرج وتُنفَّذ الحلقة 0 مرات.
    \item حلقة \textbf{نهائية}: المخرج \textenglish{\texttt{5}} وتُنفَّذ الحلقة مرة واحدة، لأنّ قيمة \textenglish{i} تصبح \textenglish{\texttt{0.5}} بعد القسمة فينتفي الشرط.
    \item حلقة \textbf{لا نهائية}: قيمة \textenglish{i} تبقى دائمًا 5 لأنّ \textenglish{\texttt{i \% 10 = 5}}، ولذلك يبقى الشرط محققًا دائمًا.
\end{enumerate}
\end{boxSolution}
\fi

\end{english}
