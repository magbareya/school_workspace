\item أمامكم عملية \textbf{تستقبل} من المستخدم \textbf{سعر} صندوق البيتسا و\textbf{كمية} الصناديق المطلوبة. تحسب العملية وتطبع المبلغ \textbf{الكلي} الذي يجب دفعه.

أكملوا الناقص في مقطع البرنامج التالي باستخدام مخزن الكلمات.

\begin{boxCode}
\begin{english}
\begin{minted}{python}
def calculate_pizza_order():
    # Read input from the user
    num_pizzas = ______________________________
    ________________ = float(input("Enter the price per pizza: "))

    # Calculate the total price
    total_price = num_pizzas * price_per_pizza

    # Display the result
    print("The total price for pizzas is:", ____________, "nis")

# Call the function
calculate_pizza_order()
\end{minted}
\end{english}
\end{boxCode}

\textbf{مخزن الكلمات:}
\begin{center}
\begin{english}
\begin{boxCode}
\texttt{int(input("Enter the number of pizzas: "))} \quad , \quad \texttt{price\_per\_pizza} \quad , \quad \texttt{total\_price}
\end{boxCode}
\end{english}
\end{center}

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def calculate_pizza_order():
    # Read input from the user
    num_pizzas = int(input("Enter the number of pizzas: "))
    price_per_pizza = float(input("Enter the price per pizza: "))

    # Calculate the total price
    total_price = num_pizzas * price_per_pizza

    # Display the result
    print("The total price for pizzas is:", total_price, "nis")

# Call the function
calculate_pizza_order()
\end{minted}
\end{english}
\end{boxSolution}

\fi

% ---------------------------------------------------------------- 3
\item أمامكم عملية \textbf{تطبع} معلومات عن المدرسة:

\begin{boxCode}
\begin{english}
\begin{minted}{python}
def school_info():
    school_name = "ABC High School"
    address = "123 Main St, Anytown"
    principal = "Mrs. Smith"
    print("School:", school_name)
    print("Address:", address)
    print("Principal:", principal)

# Call the function
school_info()
\end{minted}
\end{english}
\end{boxCode}

\begin{enumerate}[leftmargin=*, itemsep=1.2em]
\item[\textbf{أ.}] اختاروا الإجابة \textbf{الأكثر صحة}: لكي \textbf{نستقبل} اسم المدرسة من المستخدم علينا كتابة الأمر التالي:
\begin{boxCode}
\begin{english}
\begin{minted}{text}
a. input("Enter school name: ")
b. int(input("Enter school name: "))
c. float(input("Enter school name: "))
d. print("Enter school name: ")
\end{minted}
\end{english}
\end{boxCode}

\item[\textbf{ب.}] اختاروا الإجابة \textbf{الأكثر صحة}: اسم العملية هو:
\begin{boxCode}
\begin{english}
\begin{minted}{text}
a. school_info
b. principal
c. school_name
d. def
\end{minted}
\end{english}
\end{boxCode}

\item[\textbf{ج.}] عدّلوا العملية بحيث \textbf{تستقبل} من المستخدم المعلومات (اسم المدرسة، العنوان، واسم المديرة) و\textbf{تطبع} رسالة ملائمة.
\end{enumerate}

\ifwithsols
\begin{boxSolution}[البندان أ و ب]
\textbf{أ.} الإجابة الصحيحة: \textenglish{a}. \quad \textbf{ب.} الإجابة الصحيحة: \textenglish{a}.
\end{boxSolution}

\begin{boxSolution}[البند ج]
\begin{english}
\begin{minted}{python}
def school_info():
    school_name = input("Enter school name: ")
    address = input("Enter address: ")
    principal = input("Enter principal name: ")
    print("School:", school_name)
    print("Address:", address)
    print("Principal:", principal)

# Call the function
school_info()
\end{minted}
\end{english}
\end{boxSolution}

\fi

\end{enumerate}

\section*{القسم الثاني: كتابة عمليات تتلقى معطيات}

\begin{enumerate}[resume, label=\textbf{تمرين \arabic*:}, wide=0pt, leftmargin=*, itemsep=2.2em]

% ---------------------------------------------------------------- 4
\item اكتبوا عملية باسم \textenglish{rectangle()} \textbf{تتلقى} معطيين صحيحين \textenglish{a} و \textenglish{b} يمثلان طولي ضلعين متجاورين في مستطيل. على العملية أن \textbf{تحسب} و\textbf{تطبع} \textbf{مساحة} المستطيل.

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def rectangle(a, b):
    area = a * b
    print("The area of the rectangle is:", area)

# Call the function
rectangle(4, 6)
\end{minted}
\end{english}
\end{boxSolution}

\filbreak
\fi

% ---------------------------------------------------------------- 5
\item اكتبوا عملية باسم \textenglish{pool()} \textbf{تتلقى} معطيين: عدد المرات التي نريد الذهاب بها إلى البركة (\textenglish{times}) وسعر تذكرة الدخول (\textenglish{price}). تحسب العملية و\textbf{تطبع} المبلغ \textbf{الكلي} الذي يجب دفعه.

% TODO: verify this sentence and its numbers against the original PDF (the extracted text was garbled).
\textbf{احسبوا بدون برمجة:} إذا كان سعر الاشتراك للبركة 200 ش.ج، وسعر تذكرة الدخول للمرة الواحدة 20 ش.ج، فهل من الأفضل شراء اشتراك أم دفع ثمن تذكرة في كل مرة؟

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def pool(times, price):
    total = times * price
    print("The total price is:", total)

# Call the function
pool(3, 20)
\end{minted}
\end{english}
\end{boxSolution}

\filbreak
\fi

% ---------------------------------------------------------------- 6
\item اكتبوا عملية باسم \textenglish{super\_market()} \textbf{تتلقى} ثلاثة معطيات:

\begin{itemize}[itemsep=0.3em]
\item \textenglish{juice\_bottle}: عدد قناني العصير.
\item \textenglish{bottle\_price}: سعر القنينة الواحدة.
\item \textenglish{discount}: التخفيض. (افترضوا أنّ التخفيض نسبة مئوية، مثلًا \textenglish{20} تعني \textenglish{20\%}.)
\end{itemize}

تحسب العملية و\textbf{تطبع} السعر الذي يجب دفعه لشراء قناني العصير \textbf{قبل} التخفيض و\textbf{بعده}، مع رسالة ملائمة لكل سعر.

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def super_market(juice_bottle, bottle_price, discount):
    price_before = juice_bottle * bottle_price
    price_after = price_before - price_before * discount / 100
    print("Price before discount:", price_before)
    print("Price after discount:", price_after)

# Call the function
super_market(10, 5, 20)
\end{minted}
\end{english}
\end{boxSolution}

\filbreak
\fi

% ---------------------------------------------------------------- 7
\item اكتبوا عملية باسم \textenglish{ice\_cream()} \textbf{تتلقى} معطيين: طعم البوظة وتقييمها (عدد صحيح من 1 إلى 5). \textbf{تطبع} العملية رسالة تعرض طعم البوظة وتقييمها.

استدعوا العملية \textbf{3 مرات} مع قيم مختلفة للمعطيات.

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def ice_cream(flavor, rating):
    print("The", flavor, "ice cream got a rating of", rating, "out of 5")

# Call the function
ice_cream("Vanilla", 5)
ice_cream("Chocolate", 4)
ice_cream("Pistachio", 3)
\end{minted}
\end{english}
\end{boxSolution}

\fi

\filbreak
% ---------------------------------------------------------------- 8
\item اكتبوا عملية باسم \textenglish{certificate()} \textbf{تتلقى} المعطيين التاليين:

\begin{itemize}[itemsep=0.3em]
\item \textenglish{student\_name}: اسم الطالب/ة.
\item \textenglish{message}: رسالة.
\end{itemize}

على العملية أن \textbf{تطبع} شهادة إنهاء للطالب تهنّئه على نجاحه في دراسة موضوع ``العمليات''.

\ifdetailed
\begin{boxExample}
\begin{english}
\begin{minted}{text}
*****************************
**** Well Done! ****
**** Alice ****
**** you know functions! ****
*****************************
\end{minted}
\end{english}
\end{boxExample}
\fi

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
def certificate(student_name, message):
    print("*****************************")
    print("**** Well Done! ****")
    print("****", student_name, "****")
    print("****", message, "****")
    print("*****************************")

# Call the function
certificate("Alice", "you know functions!")
\end{minted}
\end{english}
\end{boxSolution}
\fi

\end{enumerate}
