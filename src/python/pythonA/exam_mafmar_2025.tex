\section*{الفصل الثاني — اختياري ($40$ علامة)}
في هذا الفصل $4$ أسئلة. يجب الإجابة عن \textbf{سؤالين} من بين الأسئلة $6$–$9$ (لكل سؤال — $20$ علامة).

\begin{enumerate}[resume, label=\textbf{السؤال \arabic*:}, wide=0pt, itemsep=2.2em]

% ---------------------------------------------------------------
\item ($20$ علامة) أمامكم مقطع برنامج يستقبل عددين صحيحين موجبين وينفّذ حسابًا معيّنًا:

\begin{boxCode}
\begin{english}
\begin{minted}{python}
num1 = int(input('Please enter number '))
num2 = int(input('Please enter number '))
while num1 >= num2:
    num1 = num1 - num2
print(num1)
\end{minted}
\end{english}
\end{boxCode}

\begin{enumerate}[label=\alph*.]
\item أكملوا الجدول التالي لقيم \textenglish{num1} و \textenglish{num2} المعطاة:

{\begin{english}
\renewcommand{\arraystretch}{1.4}
\begin{longtable}{|C{3cm}|C{3cm}|C{3.5cm}|}
\hline
\textbf{num1} & \textbf{num2} & \textarabic{\textbf{المُخرج}} \\
\hline
$11$ & $4$ & \ifwithsols $\mathbf{3}$ \fi \\ \hline
$9$ & $3$ & \ifwithsols $\mathbf{0}$ \fi \\ \hline
$5$ & $2$ & \ifwithsols $\mathbf{1}$ \fi \\ \hline
$2$ & $5$ & \ifwithsols $\mathbf{2}$ \fi \\ \hline
\end{longtable}
\end{english}}
\filbreak

\item ماذا ينفّذ مقطع البرنامج؟ أحيطوا بدائرة الإجابة الصحيحة:
\begin{enumerate}[label=\arabic*.]
\item يحسب ويطبع حاصل ضرب \textenglish{num1} بـ \textenglish{num2}.
\item يحسب ويطبع الفرق بين \textenglish{num1} و \textenglish{num2}.
\item يحسب ويطبع باقي القسمة (المودولو) من قسمة \textenglish{num1} على \textenglish{num2}.
\item يحسب ويطبع القوّة لـ \textenglish{num1} بالأسّ \textenglish{num2}.
\end{enumerate}

\ifwithsols
\begin{boxSolution}
الإجابة الصحيحة هي \textbf{$3$}: يحسب ويطبع باقي القسمة (المودولو) من قسمة \textenglish{num1} على \textenglish{num2} عن طريق الطرح المتكرّر لـ \textenglish{num2} من \textenglish{num1} طالما أنّ $\text{num1} \geq \text{num2}$.
\end{boxSolution}
\fi
\filbreak
\end{enumerate}
\filbreak

\begin{enumerate}[label=\alph*.]

\item ($20$ علامة) \textbf{النص "المضاعَف زوجيًا"} هو نص جديد يُبنى من نص موجود، بحيث نضيف الرمز \textenglish{\texttt{'\^{}'}} بعد كل حرف يقع في مكان (مؤشر) \textbf{زوجي} في النص الأصلي.
\begin{itemize}
\item \textbf{مثال $1$:} للنص \textenglish{\texttt{"Monkey"}}، ينتج النص المضاعَف زوجيًا التالي: \textenglish{\texttt{"M\^{}on\^{}ke\^{}y"}}.
\item \textbf{مثال $2$:} للنص \textenglish{\texttt{"Tiger"}}، ينتج النص المضاعَف زوجيًا التالي: \textenglish{\texttt{"T\^{}ig\^{}er\^{}"}}.
\end{itemize}
العملية التي أمامكم تستقبل نصًا \textenglish{msg} من المستخدم، وتعرض نصًا "مضاعفًا زوجيًا". أكملوا الفراغات $(1)$–$(5)$ في العملية:

\item ($20$ علامة) \textbf{العدد "الموقِظ"} هو عدد صحيح موجب يكون فيه مجموع جميع منازله \textbf{أكبر بـ $3$ أضعاف أو أكثر} من عدد المنازل في العدد.
\begin{itemize}
\item \textbf{مثال $1$:} العدد $129$ هو عدد "موقِظ" لأنّ مجموع منازله ($1 + 2 + 9 = 12$) أكبر بـ $3$ أضعاف أو أكثر من عدد منازله ($3$). الحساب: $12 / 3 \geq 3$.
\item \textbf{مثال $2$:} العدد $5$ هو عدد "موقِظ" لأنّ مجموع منازله ($5$) أكبر بـ $3$ أضعاف أو أكثر من عدد منازله ($1$). الحساب: $5 / 1 \geq 3$.
\item \textbf{مثال $3$:} العدد $124$ ليس عددًا "موقِظًا" لأنّ مجموع منازله ($1 + 2 + 4 = 7$) ليس أكبر بـ $3$ أضعاف أو أكثر من عدد منازله ($3$). الحساب: $7 / 3 < 3$.
\end{itemize}

\begin{enumerate}[label=\alph*.]
\item ($4$ علامات) حسب التعريف أعلاه، هل $4158$ هو عدد "موقِظ"؟ إذا نعم، اعرضوا حسابًا رياضيًا (كما في المثال) يُظهر ذلك؛ وإلّا اشرحوا.
\ifwithsols
\begin{boxSolution}
\textbf{نعم}، العدد $4158$ هو عدد "موقِظ" لأنّ مجموع منازله هو $4 + 1 + 5 + 8 = 18$، وعدد منازله هو $4$، ويتحقّق:
$$18 / 4 = 4.5 \geq 3 \quad (\text{أو } 18 \geq 3 \cdot 4 = 12)$$
\end{boxSolution}
