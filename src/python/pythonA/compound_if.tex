\section*{كتابة برنامج - اختياري}
\begin{enumerate}[itemsep=3em]

\item \textbf{برنامج 1: شراء حاسوب نقال} \\
اكتبوا برنامج يستقبل المبلغ المالي الذي تم توفيره وعدد أشهر التوفير. إذا كان المبلغ الذي تم توفيره أكثر من 4000 ش.ج أو عدد الاشهر أكثر من 10 اطبعوا عبارة توضيحية أنه "أصبح باستطاعتكم شراء حاسوب نقال".

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
savings = float(input("أدخل المبلغ المالي الموفر (ش.ج): "))
months = int(input("أدخل عدد أشهر التوفير: "))

if savings > 4000 or months > 10:
    print("أصبح باستطاعتكم شراء حاسوب نقال")
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item \textbf{برنامج 2: مساحة الذاكرة المتاحة} \\
اكتبوا برنامج يستقبل مساحة الذاكرة المتاحة في مجلّد الصور المحوسب. إذا كانت القيمة المُدخلة اقل من 2 ميجابايت (\textenglish{MB})، اطبعوا عبارة توضيحية تنص أن المكان امتلأ ويجب حذف صور.

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
free_space_mb = float(input("أدخل مساحة الذاكرة المتاحة بالميجابايت (MB): "))

if free_space_mb < 2:
    print("المكان امتلأ ويجب حذف صور")
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item \textbf{برنامج 3: قطار الجبال} \\
اكتبوا برنامج يستقبل طول وعمر شخص. إذا كان طول الشخص 130 سم وما فوق وعمره 13 سنة وما فوق، اطبعوا عبارة توضيحية أن بامكانه الدخول لقطار الجبال.

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
height = float(input("أدخل طول الشخص (سم): "))
age = int(input("أدخل عمر الشخص (سنوات): "))

if height >= 130 and age >= 13:
    print("بإمكانك الدخول لقطار الجبال")
\end{minted}
\end{english}
\end{boxSolution}
\fi

\item \textbf{برنامج 4: رخصة القيادة} \\
يحصل الطالب على رخصة قيادة إذا تحققت الشروط التالية: عمره فوق 17، نجح في امتحان السياقة النظري (تئوريا) ونجح في امتحان السياقة العملي. اكتبوا برنامج يستقبل المتطلبات الثلاثة وفي حالة أن الطالب لا يستحق رخصة سياقة، يقوم بطباعة السبب.

\ifwithsols
\begin{boxSolution}
\begin{english}
\begin{minted}{python}
age = float(input("أدخل عمر الطالب: "))
passed_theory = input("هل نجحت في الامتحان النظري؟ (نعم/لا): ")
passed_practical = input("هل نجحت في الامتحان العملي؟ (نعم/لا): ")

reasons = []
if age <= 17:
    reasons.append("العمر يجب أن يكون فوق 17 سنة")
if passed_theory != "نعم":
    reasons.append("لم تنجح في امتحان السياقة النظري")
if passed_practical != "نعم":
    reasons.append("لم تنجح في امتحان السياقة العملي")

if len(reasons) == 0:
    print("تهانينا! أنت تستحق الحصول على رخصة القيادة.")
else:
    print("لا تستحق رخصة القيادة للأسباب التالية:")
    for reason in reasons:
        print("- ", reason)
\end{minted}
\end{english}
\end{boxSolution}
\fi

\end{enumerate}
