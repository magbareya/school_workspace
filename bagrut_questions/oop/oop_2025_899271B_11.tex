\begin{english}
\begin{minted}{csharp}
public double TotalPrice()
{
    int compCount = 0, gardenCount = 0;
    for (int i = 0; i < this.current; i++)
    {
        if (this.chairs[i] is ComputerChair)
        {
            compCount++;
        }
        else if (this.chairs[i] is GardenChair)
        {
            gardenCount++;
        }
    }

    double total = 0;
    for (int i = 0; i < this.current; i++)
    {
        Chair c = this.chairs[i];
        Console.WriteLine("Price: " + c.GetPrice() +
            ", Date: " + c.GetCreateDate().GetMonth() + "/" + c.GetCreateDate().GetYear());

        if (c is ComputerChair)
        {
            ComputerChair cc = (ComputerChair)c;
            Console.WriteLine("Wheels: " + cc.GetWheels() + ", Height: " + cc.GetAdjustHeight());
            total += (compCount > 10) ? c.GetPrice() * 0.8 : c.GetPrice();
        }
        else if (c is GardenChair)
        {
            GardenChair gc = (GardenChair)c;
            Console.WriteLine("Foldable: " + gc.GetIsFoldable());
            total += (gardenCount > 15) ? c.GetPrice() * 0.75 : c.GetPrice();
        }
        else if (c is BarStool)
        {
            BarStool bs = (BarStool)c;
            Console.WriteLine("Height: " + bs.GetAdjustHeight());
            total += c.GetPrice();
        }
    }
    return total;
}
\end{minted}
\end{english}
\end{boxCode}
