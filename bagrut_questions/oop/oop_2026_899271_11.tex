    \item فحص المخطّطات الهرميّة $1$–$6$:
    \begin{itemize}[itemsep=0.25em]
        \item \textbf{مخطّط $1$ — لا يمكن:} الفئة \textenglish{A} ترث من \textenglish{B} و\textenglish{C}، لذا لا يمكن إسناد \textenglish{new C()} أو \textenglish{new B()} إلى متغيّر من نمط \textenglish{A}.
        \item \textbf{مخطّط $2$ — يمكن.}
        \item \textbf{مخطّط $3$ — لا يمكن:} الفئة \textenglish{B} لا ترث من \textenglish{C} (\textenglish{C c1 = new B()} غير سليم)، و\textenglish{D} لا ترث من \textenglish{C}، و\textenglish{C} لا ترث من \textenglish{A}.
        \item \textbf{مخطّط $4$ — يمكن.}
        \item \textbf{مخطّط $5$ — لا يمكن:} الفئة \textenglish{B} فوق \textenglish{A} و\textenglish{C}، لذا لا يصحّ \textenglish{C c1 = new B()} ولا \textenglish{A a2 = new B()}.
        \item \textbf{مخطّط $6$ — يمكن.}
    \end{itemize}

    \item \textbf{نعم، تتغيّر الإجابة:} لكي يعمل التحويل \textenglish{D d1 = (D)a3;} (حيث الكائن الفعلي من نمط \textenglish{B}) دون خطأ في زمن التشغيل، يجب أن تكون \textenglish{B} وريثة لـ \textenglish{D}. وبذلك يصبح \textbf{المخطّط $6$ فقط} هو الممكن (المخطّطان $2$ و$4$ غير ممكنين).
\end{enumerate}
