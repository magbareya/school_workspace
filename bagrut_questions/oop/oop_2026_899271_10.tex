\begin{minted}{csharp}
public int WithSticks()
{
    int res = 0;
    for (int i = 0; i < this.count; i++)
    {
        Instrum ins = this.arrM[i].GetInstrum();
        if (ins is Perc && ((Perc)ins).GetHasSticks())
        {
            res++;
        }
    }
    return res;
}
\end{minted}
\end{english}

\begin{english}
\begin{minted}{csharp}
public bool Insert(Musician musician)
{
    if (this.count == this.arrM.Length)
    {
        return false;
    }

    int pos = this.count;
    Instrum ins = musician.GetInstrum();
    if (ins is Bow)
    {
        pos = 0;
    }
    else if (ins is Wind)
    {
        pos = 0;
        while (pos < this.count && !(this.arrM[pos].GetInstrum() is Perc))
        {
            pos++;
        }
    }

    for (int i = this.count; i > pos; i--)
    {
        this.arrM[i] = this.arrM[i - 1];
    }
    this.arrM[pos] = musician;
    this.count++;
    return true;
}
\end{minted}
\end{english}
\end{boxCode}
