public double AvgRate(int type)
{
    int sum = 0;
    int cnt = 0;
    for (int i = 0; i < this.count; i++)
    {
        if (type == 1 && this.games[i] is VRGame)
        {
            sum += this.games[i].GetRate();
            cnt++;
        }
        else if (type == 2 && this.games[i] is ComputerGame && !(this.games[i] is VRGame))
        {
            sum += this.games[i].GetRate();
            cnt++;
        }
    }
    return (double)sum / cnt;
}
\end{minted}
\end{english}
