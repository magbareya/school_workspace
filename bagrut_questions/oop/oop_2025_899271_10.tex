public class Bird : Animal

public Bird(string color, int weight, bool isFlying) : base(color, weight)

public Fish(string color, int weight, string waterType) : base(color, weight)

    this.waterType = waterType;

public Snake(string color, int weight, int length, bool isVenomous)

    this.length = length;

\begin{boxSolution}[\texorpdfstring{سؤال $10$ امتحان $\mathrm{899271}$ سنة $2025$ - فرع جـ}{سؤال 10 امتحان 899271 سنة 2025 - فرع جـ}]
\textbf{جـ.}
\begin{enumerate}[label=\textbf{(\arabic*)}, itemsep=0.5em]
    \item نضيف \textenglish{ToString} في الفئات \textenglish{Bird} و \textenglish{Parrot} و \textenglish{Snake} فقط (الفئة \textenglish{Fish} ترث \textenglish{Animal.ToString()} كما هي):
\begin{boxCode}
\begin{english}
\begin{minted}{csharp}
// In class Bird:
public override string ToString()
{
    return "I'm a bird! " + base.ToString();
}

// In class Parrot:
public override string ToString()
{
    return "I'm a parrot! My color is " + this.GetColor() + "!";
}

// In class Snake:
public override string ToString()
{
    if (this.isVenomous)
    {
        return "I'm a snake! I'm venomous, be careful! " + base.ToString();
    }
    return "I'm a snake! I'm not venomous! " + base.ToString();
}
\end{minted}
\end{english}
\end{boxCode}

    \item العملية \textenglish{Hello} غير سليمة (خطأ ترجمة)، لأنّ النمط الساكن لـ \textenglish{arr[i]} هو \textenglish{Animal} ولا توجد عملية \textenglish{Zoo()} في الفئة \textenglish{Animal}. التصحيح:
\begin{boxCode}
\begin{english}
\begin{minted}{csharp}
public static void Hello(Animal[] arr)
{
    for (int i = 0; i < arr.Length; i++)
    {
        if (arr[i] is Bird)
        {
            ((Bird)arr[i]).Zoo();
        }
    }
}
\end{minted}
\end{english}
\end{boxCode}
    المُخرج بالنسبة للمصفوفة \textenglish{animals}: تطبع \textenglish{Hello} سطرين (مرّة لـ \textenglish{Bird} ومرّة لـ \textenglish{Parrot}).
\end{enumerate}
\end{boxSolution}
