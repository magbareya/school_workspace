\begin{enumerate}[label=\textbf{(\arabic*)}, itemsep=0.6em]
    \item العملية البنائية للفئة \textenglish{Vehicle}:
\begin{boxCode}
\begin{english}
\begin{minted}{csharp}
public Vehicle(string name, int days, int kilo, string id)
{
    this.id = id;
    this.contract = new Contract(name, days, kilo);
}
\end{minted}
\end{english}
\end{boxCode}

    \item العملية البنائية للفئة \textenglish{Car}:
\begin{boxCode}
\begin{english}
\begin{minted}{csharp}
public Car(string name, int days, int kilo, string id, int seats)
    : base(name, days, kilo, id)
{
    this.seats = seats;
}
\end{minted}
\end{english}
\end{boxCode}
\end{enumerate}
\end{boxSolution}

    return base.Payment() + 500;

    return base.Payment() / 2.0;
}
\end{minted}
\end{english}
\end{boxCode}
