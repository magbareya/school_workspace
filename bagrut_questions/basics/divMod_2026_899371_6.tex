\begin{boxSolution}[سؤال 6 امتحان 899371 سنة 2026]
في البداية نعرّف عملية مساعدة تحسب عدد خانات العدد المتلقى:

\begin{boxCode}
\begin{english}
\begin{minted}{csharp}
public static int GetDigitsCount(int num)
{
    int len = 0;
    while (num > 0)
    {
        len++;
        num /= 10;
    }
    return len;
}
\end{minted}
\end{english}
\end{boxCode}


\begin{enumerate}[label=\alph*.]
    \item .
\begin{boxCode}
\begin{english}
\begin{minted}{csharp}
public static int LeftDigits(int num, int k)
{
    int len = GetDigitsCount(num);
    return num / (int)Math.Pow(10, len - k);
}
\end{minted}
\end{english}
\end{boxCode}

    int k = len / 2;
    int left = LeftDigits(squared, k);
