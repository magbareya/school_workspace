public static void Guess(string secretStr, string playerStr)
{
    if (secretStr.Length != playerStr.Length)

        Console.WriteLine("Size Error");
        return;
    }

    int dakik = 0, isaba = 0, tadyea = 0;
    for (int i = 0; i < playerStr.Length; i++)
    {
        bool isFound = False;
        for(int j = 0; j < secretStr.Length; j++)
        {
            if(playerStr[i] == secretStr[j])
            {
                isFound = true;
                if(i == j)
                    dakik++;
                else
                    isaba++;
            }
        }
        if(!isFound)
            tadyea++;
    }
    Console.WriteLine(dakik);
    Console.WriteLine(isaba);
    Console.WriteLine(tadyea);
}
